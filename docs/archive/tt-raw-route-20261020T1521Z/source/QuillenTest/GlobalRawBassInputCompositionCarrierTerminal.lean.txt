Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassInputCompositionCarrierTerminal

/-! Regression of the complete terminal surface against all four original carrier regions. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
open CategoryTheory AlgebraicGeometry Quillen.LoopPrespectrum
open Quillen.Waldhausen.IteratedS Quillen.NativePathCells
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open Quillen.PerfectComplexes Quillen.PerfectComplexes.GlobalRawBass
universe u
namespace QuillenTest.GlobalRawBassInputCompositionCarrierTerminal
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- All four original parameter regions retain their full carrier values simultaneously. -/
theorem original_regions (s t : unitInterval) :
    (InputCompositionCarrierTerminal.face f g (lower s, t),
      InputCompositionCarrierTerminal.face f g (upper s, lower t),
      InputCompositionCarrierTerminal.face f g (upper s, upper (lower t)),
      InputCompositionCarrierTerminal.face f g (upper s, upper (upper t))) =
    (InputCompositionTerminalAttach.postFace f g (s, t),
      InputCompositionTerminalProducts.face f g plus plus_projection (s, t),
      InputCompositionTerminalAttach.preFace f g (s, t),
      InputCompositionTerminalProducts.face f g minus minus_projection
        (s, unitInterval.symm t)) := by
  simp only [InputCompositionCarrierTerminal.face_lower,
    InputCompositionCarrierTerminal.face_upper, InputCompositionTerminalSecond.face_plus,
    InputCompositionTerminalSecond.face_middle, InputCompositionTerminalSecond.face_minus]

/-- The two original source parameterizations agree as entire functions into the carrier. -/
theorem entire_seam :
    (fun t ↦ InputCompositionTerminalAttach.postFace f g (1, t)) =
      (fun t ↦ InputCompositionTerminalSecond.face f g (0, t)) := by
  funext t
  exact InputCompositionCarrierTerminal.seam f g t

/-- Whole native evaluation agrees as a function of both parameters and the original input. -/
theorem entire_native (n : ℕ) :
    (fun (z : unitInterval × unitInterval)
      (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) ↦
      CommonReferenceNerve.compactPrespectrumFamily (InputCompositionCarrierCharts.source f g)
        (InputCompositionCarrierTerminal.face f g) n (z, x)) =
    (fun z x ↦ (InputCompositionCharts.compositeComparison f g).level n
      (z.1, HomotopyPushout.cylinder (boundedFlatPerfectPrespectrumMap (projection.app Z))
        (boundedFlatPerfectPrespectrumMap (projection.app Z)) n (z.2, x))) := by
  funext z x
  exact InputCompositionCarrierTerminal.face_native f g n z.1 z.2 x

/-- Structure time is independent even where both original overlap subregions meet. -/
theorem structure_at_middle (n : ℕ) (s t z : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CommonReferenceNerve.compactPrespectrumFamily (InputCompositionCarrierCharts.source f g)
      (InputCompositionCarrierTerminal.face f g) (n + 1)
        ((upper s, upper (lower t)),
          (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).structureMap n x z) =
      (originalPrespectrum (boundedFlatPerfectWaldhausen (laurent.obj X))).structureMap n
        ((InputCompositionCharts.compositeComparison f g).level n
          (upper s, HomotopyPushout.cylinder
            (boundedFlatPerfectPrespectrumMap (projection.app Z))
            (boundedFlatPerfectPrespectrumMap (projection.app Z)) n (upper (lower t), x))) z := by
  rw [InputCompositionCarrierTerminal.face_structure,
    InputCompositionCarrierTerminal.face_native]

end QuillenTest.GlobalRawBassInputCompositionCarrierTerminal
