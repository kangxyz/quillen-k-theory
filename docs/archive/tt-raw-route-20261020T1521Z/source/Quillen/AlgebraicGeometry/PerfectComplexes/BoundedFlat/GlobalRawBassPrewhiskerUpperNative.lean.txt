variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c

/-- The entire original upper-span input word is recovered without changing any of its clocks. -/
theorem upperSpan_native (n : ℕ) (r s t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen W)).space n) :
    ProjectionReference.FourEdge.coneEval X (A ≫ E) n
      (upperSpan f g c ((r, s), t)) x =
    (InputComposition.comparisonConeNullSquare f g).level n
      (r, (s, (spanMap c).map n
        (HomotopyPushout.cylinder (boundedFlatPerfectPrespectrumMap (projection.app W))
          (boundedFlatPerfectPrespectrumMap (projection.app W)) n (t, x)))) := by
  rcases cover t with ⟨v, rfl⟩ | ⟨v, rfl⟩
  · rw [upperSpan_lower, spanMap_cylinder]
    simp only [HomotopyPushout.mapSquareHomotopy, Homotopy.trans_lower_cell]
    exact (selectedPlus_original f g c n r s v x).trans
      (InputCompositionPositive.nullSquare_factor f g n r s _)
  · rcases cover v with ⟨w, rfl⟩ | ⟨w, rfl⟩
    · rw [upperSpan_middle, spanMap_cylinder]
      simp only [HomotopyPushout.mapSquareHomotopy, Homotopy.trans_upper_cell,
        Homotopy.trans_lower_cell]
      exact (ProjectionReference.FourEdge.Precomp.cone_native X A E n
        (InputCompositionOverlapScalar.correctedFamily f g ((r, s), w)) x).trans
        (InputCompositionOverlapScalar.correctedSquare_native f g n w
          ((prespectrumMap A).map n x) r s)
    · rw [upperSpan_upper, spanMap_cylinder]
      simp only [HomotopyPushout.mapSquareHomotopy, Homotopy.trans_upper_cell]
      exact (selectedMinus_original f g c n r s (unitInterval.symm w) x).trans
        (InputCompositionNegative.nullSquare_factor f g n r s _)
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
