/-- Polynomial target nodes for the exact original reference and parameter data. -/
theorem polynomial_target_nodes (i : Fin 3) :
    InputCompositionSpanScalar.polynomialNode f g
      (InputCompositionTopPieces.polynomialNodes f g (CoefficientMovingSquareRows.targetSide i)) =
    polynomialNode f g (InputCompositionPolynomialScalar.polynomialNodes f g i) := by
  have h := CoefficientMovingSquareRows.target_side_nodes
    g f (polynomial.map g) (polynomial.map f)
    (projection.app Z) (projection.app Y) (projection.app X)
    (projection.naturality g).symm (projection.naturality f).symm i
  refine (congrArg (InputCompositionSpanScalar.polynomialNode f g) h).trans ?_
  simp only [InputCompositionSpanScalar.polynomialNode, reanchor, polynomialNode,
    polynomialAnchor, BassH0Merge.preReferenceNode, precompNode, Iso.trans_assoc,
    InputCompositionTopChartRows.movingBaseAnchor,
    InputCompositionPolynomialScalar.polynomialNodes,
    InputCompositionPolynomialScalar.polynomialReference, BassH0Merge.binaryNodes,
    InputCompositionCarrierCharts.projectionMap, BassH0Merge.normalizedMap]

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionProjectionAttachment
