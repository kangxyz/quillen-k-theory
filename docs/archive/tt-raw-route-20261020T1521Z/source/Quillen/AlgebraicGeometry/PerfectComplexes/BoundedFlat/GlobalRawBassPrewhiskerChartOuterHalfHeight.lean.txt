/-! The two outer pieces of the actual half-height gluing retain their original clocks. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Quillen.Waldhausen.IteratedS
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open Quillen.SimplexParameters Quillen.NativePathCells
open scoped Simplicial Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
open CommonReferenceNerve
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "J" => ProjectionReference.projectionMap X
local notation "L" => ProjectionReference.reference X (A ≫ E)
local notation "β" => ProjectionReference.FourEdge.beta X (A ≫ E)

/-- The first original half-height segment is exactly the selected positive branch bottom. -/
theorem outer_halfHeight_plus (r t : I) :
    FourEdgeTree.laurentInclusion (A ≫ E) J L β
      (anchoredLowerLaurent f g c ((r, 1), lower t)) =
        selectedPlus f g c ((r, 0), t) := by
  rw [selectedPlus_bottom]
  change FourEdgeTree.laurentInclusion (A ≫ E) J L β
    (mapParameter _ (FinitePrecomp.parameter _ _ _
      (lowerLaurentPath f g r, BassH0Diagram.parameter (1, lower t)))) = _
  rw [BassH0Diagram.parameter_one, path_lower]
  exact congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L β)
    (outer_plus_scalar_parameter f g c r t)

/-- The last original quarter retains the reversed negative clock, without replacing the path. -/
theorem outer_halfHeight_minus (r t : I) :
    FourEdgeTree.laurentInclusion (A ≫ E) J L β
      (anchoredLowerLaurent f g c ((r, 1), upper (upper t))) =
        selectedMinus f g c ((r, 0), unitInterval.symm t) := by
  rw [selectedMinus_bottom]
  change FourEdgeTree.laurentInclusion (A ≫ E) J L β
    (mapParameter _ (FinitePrecomp.parameter _ _ _
      (lowerLaurentPath f g r, BassH0Diagram.parameter (1, upper (upper t))))) = _
  rw [BassH0Diagram.parameter_one, path_upper, path_upper]
  exact congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L β)
    (outer_minus_scalar_parameter f g c r (unitInterval.symm t))

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
