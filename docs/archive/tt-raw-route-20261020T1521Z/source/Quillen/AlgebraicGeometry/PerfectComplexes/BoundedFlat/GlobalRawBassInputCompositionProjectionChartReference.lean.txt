/-- Laurent Node for the exact original reference and parameter data. -/
def laurentNode (N : Node G) : Node L :=
  reanchor (InputCompositionTopPieces.middleAnchor f g).symm
    (InputCompositionPolynomialOverlap.overlapNode f g k hk N)

/-- Side Anchor eq for the exact original reference and parameter data. -/
theorem sideAnchor_eq : InputCompositionTopChartRows.movingSideAnchor f g k hk =
    InputCompositionTopChartRows.chartSideAnchor f g k hk := by
  have h := (InputCompositionTopChartRows.moving_row f g k hk 0).symm.trans
    ((InputCompositionTopChartRows.row_nodes f g k hk 0).trans
      (InputCompositionTopChartRows.chart_row f g k hk 0))
  exact prePost_anchor_eq_of_node (InputCompositionCarrierCharts.source f g) D K
    (InputCompositionTopChartRows.binarySource f g)
    (InputCompositionTopChartRows.binaryFamily f g) 0 h

/-- Attachment anchor for the exact original reference and parameter data. -/
theorem attachment_anchor :
    (InputCompositionTopPieces.middleAnchor f g ≪≫ ProjectionReference.chartAnchor X k hk E) ≪≫
      isoWhiskerRight (polynomialAnchor f g) (K).functor =
    InputCompositionCarrierCharts.chartReferenceIso f g k hk ≪≫
      isoWhiskerLeft (D).functor ε := by
  exact (congrArg
    (fun e ↦ (InputCompositionTopPieces.middleAnchor f g ≪≫
      ProjectionReference.chartAnchor X k hk E) ≪≫ e)
    (isoWhiskerRight_trans (InputCompositionSpanScalar.polynomialAnchor f g).symm
      (InputCompositionTopChartRows.movingBaseAnchor f g) (K).functor)).trans
    ((Iso.trans_assoc _ _ _).symm.trans
      ((congrArg (fun e ↦ e ≪≫
        isoWhiskerRight (InputCompositionTopChartRows.movingBaseAnchor f g) (K).functor)
        (InputCompositionSpanScalar.moving_anchor f g k hk)).trans
          (sideAnchor_eq f g k hk)))
