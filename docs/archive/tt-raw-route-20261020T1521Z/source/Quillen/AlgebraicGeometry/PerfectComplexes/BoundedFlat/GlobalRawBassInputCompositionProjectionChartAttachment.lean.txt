/-- Chart attachment node for the exact original reference and parameter data. -/
theorem chart_attachment_node (N : Node P) :
    laurentNode f g k hk (attachPostcompNode P K G ε N) =
      ProjectionReference.chartNode X k hk E (polynomialNode f g N) := by
  have h := congrArg
    (fun e ↦ (InputCompositionTopPieces.middleAnchor f g).symm ≪≫ e)
    (attachment_anchor f g k hk)
  have hbase :
      (InputCompositionTopPieces.middleAnchor f g).symm ≪≫
        (InputCompositionCarrierCharts.chartReferenceIso f g k hk ≪≫
          isoWhiskerLeft (D).functor ε) =
      ProjectionReference.chartAnchor X k hk E ≪≫
        isoWhiskerRight (polynomialAnchor f g) (K).functor := by
    simpa only [← Iso.trans_assoc, Iso.symm_self_id, Iso.refl_trans] using h.symm
  apply Sigma.ext
  · exact Model.Hom.ext rfl
  apply heq_of_eq
  change (InputCompositionTopPieces.middleAnchor f g).symm ≪≫
    (InputCompositionCarrierCharts.chartReferenceIso f g k hk ≪≫
      isoWhiskerLeft (D).functor (ε ≪≫ isoWhiskerRight N.2 (K).functor)) =
    ProjectionReference.chartAnchor X k hk E ≪≫
      isoWhiskerRight (polynomialAnchor f g ≪≫ isoWhiskerLeft (D).functor N.2)
        (K).functor
  rw [isoWhiskerLeft_trans, isoWhiskerRight_trans, whisker_iso_interchange]
  simpa only [Iso.trans_assoc] using congrArg
    (fun e ↦ e ≪≫ isoWhiskerLeft (D).functor (isoWhiskerRight N.2 (K).functor)) hbase

/-- Laurent surface for the exact original reference and parameter data. -/
theorem laurent_surface (r s : unitInterval) :
    mapParameter (laurentNode f g k hk)
      (InputCompositionPolynomialScalar.chartSurface f g k (r, s)) =
    mapParameter (reanchor (InputCompositionTopPieces.middleAnchor f g).symm)
      (InputCompositionCarrierCharts.chartFace f g k hk (r, s)) := by
  change mapParameter _ (parameter _ _ _) = _
  rw [mapParameter_parameter, ← InputCompositionPolynomialOverlap.overlap_surface f g k hk,
    InputCompositionPolynomialScalar.chartSurface]
  change _ = mapParameter _ (mapParameter _ (parameter _ _ _))
  rw [mapParameter_parameter, mapParameter_parameter]
  rfl
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionProjectionAttachment
