private theorem attachment_node_generic {V V' W W' : Model.{u + 1}}
    (P : V ⟶ V') (H : V' ⟶ W') (J : W' ⟶ W)
    (K : V ⟶ W') (L : V' ⟶ W) (M : V ⟶ W)
    (a : K.functor ≅ (P ≫ H).functor)
    (b : L.functor ≅ (H ≫ J).functor)
    (c : M.functor ≅ (K ≫ J).functor) (N : Node H) :
    attachPrecompNode L P M
      (c ≪≫ isoWhiskerRight a J.functor ≪≫
        Functor.associator P.functor H.functor J.functor ≪≫
          isoWhiskerLeft P.functor b.symm)
      (attachPostcompNode H J L b N) =
    attachPostcompNode K J M c (attachPrecompNode H P K a N) := by
  have hF : (P ≫ (N.1 ≫ J)).functor = ((P ≫ N.1) ≫ J).functor := rfl
  apply Sigma.ext (Model.Hom.ext hF)
  apply heq_of_eq
  apply Iso.ext
  ext A
  simp only [attachPostcompNode, attachPrecompNode, isoWhiskerLeft_trans, Iso.trans_assoc,
    Iso.trans_hom, isoWhiskerRight_hom, isoWhiskerLeft_hom, Iso.symm_hom, NatTrans.comp_app,
    comp_obj, whiskerRight_app, associator_hom_app, whiskerLeft_app,
    Iso.inv_hom_id_app_assoc, isoWhiskerRight_trans, NatIso.cancel_natIso_hom_left]
  exact congrArg (fun k ↦ J.functor.map (a.hom.app A) ≫ k)
    (Category.id_comp (J.functor.map (N.2.hom.app (P.functor.obj A))))

attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- The actual normalized base premap of the second comparison. -/
abbrev premap := BassH0Merge.normalizedMap g

/-- The actual second polynomial row fixes the precomposition anchor, not an arbitrary lift. -/
def polynomialAnchor : (polyRef (f ≫ g)).functor ≅ (premap g ≫ polyRef f).functor :=
  (InputCompositionH0TopRows.outerIso f g).symm ≪≫
    CoefficientMovingSquareRows.secondAnchor g f (polynomial.map f)
      (projection.app Y) (projection.app X) (projection.naturality f).symm

/-- The Laurent anchor is forced by the two actual ordinary attachments. -/
def laurentAnchor : (laurentRef (f ≫ g)).functor ≅ (premap g ≫ laurentRef f).functor :=
  ordinary (f ≫ g) ≪≫
    isoWhiskerRight (polynomialAnchor f g) (chart (X := X)).functor ≪≫
      Functor.associator (premap g).functor (polyRef f).functor (chart (X := X)).functor ≪≫
        isoWhiskerLeft (premap g).functor (ordinary f).symm

/-- Every polynomial node retains its full reference and actual base premap. -/
def polynomialNode (N : Node (polyRef f)) : Node (polyRef (f ≫ g)) :=
  attachPrecompNode (polyRef f) (premap g) (polyRef (f ≫ g)) (polynomialAnchor f g) N

/-- Every Laurent node retains its full reference and actual base premap. -/
def laurentNode (N : Node (laurentRef f)) : Node (laurentRef (f ≫ g)) :=
  attachPrecompNode (laurentRef f) (premap g) (laurentRef (f ≫ g)) (laurentAnchor f g) N

/-- The ordinary attachment commutes on arbitrary complete nodes, before evaluation. -/
theorem attachment_node (N : Node (polyRef f)) :
    laurentNode f g (attachPostcompNode (polyRef f) (chart (X := X))
      (laurentRef f) (ordinary f) N) =
    attachPostcompNode (polyRef (f ≫ g)) (chart (X := X))
      (laurentRef (f ≫ g)) (ordinary (f ≫ g)) (polynomialNode f g N) :=
  attachment_node_generic (premap g) (polyRef f) (chart (X := X))
    (polyRef (f ≫ g)) (laurentRef f) (laurentRef (f ≫ g))
    (polynomialAnchor f g) (ordinary f) (ordinary (f ≫ g)) N


local notation "Ff" => polyRef f
local notation "Gf" => laurentRef f
local notation "J" => chart (X := X)
local notation "F" => polyRef (f ≫ g)
local notation "G" => laurentRef (f ≫ g)
local notation "εf" => ordinary f
local notation "ε" => ordinary (f ≫ g)
local notation "ε'" => shifted (f ≫ g)

/-- The complete polynomial nerve follows the fixed full-node precomposition. -/
def polynomial := mapParameter (polynomialNode f g)

/-- The complete Laurent nerve follows the matching ordinary attachment. -/
def laurent : C(|CutOpenCarrier.parameterSSet Gf|, Carrier (f ≫ g)) :=
  (CutOpenCarrier.laurentInclusion F J G ε ε').comp (mapParameter (laurentNode f g))

/-- Attachment commutes at every realized polynomial point. -/
theorem attachment_parameter (y : |CutOpenCarrier.parameterSSet Ff|) :
    mapParameter (laurentNode f g) (mapParameter (attachPostcompNode Ff J Gf εf) y) =
      mapParameter (attachPostcompNode F J G ε) (polynomial f g y) := by
  obtain ⟨k, N, z, rfl⟩ := exists_parameter Ff y
  change mapParameter _ (mapParameter _ (parameter _ _ _)) =
    mapParameter _ (mapParameter _ (parameter _ _ _))
  rw [mapParameter_parameter, mapParameter_parameter, mapParameter_parameter,
    mapParameter_parameter]
  exact congrArg (fun ns ↦ parameter G ns z) (funext (fun i ↦ attachment_node f g (N i)))

/-- The original ordinary cone tip under precomposition. -/
def tipMap : C(|CutOpenCarrier.tipSSet.{u + 1}|, Carrier (f ≫ g)) :=
  ContinuousMap.const _ (CutOpenCarrier.tipMinus F J G ε ε')

/-- The actual precomposition does not insert a new cylinder half. -/
def cylinderTrack : ((laurent f g).comp
    (SSet.toTop.map (CutOpenCarrier.attaching Ff J Gf εf)).hom).Homotopy
    ((tipMap f g).comp (SSet.toTop.map (CutOpenCarrier.erase Ff)).hom) where
  toContinuousMap := ⟨fun z ↦ CutOpenCarrier.cylinderMinus F J G ε ε'
    (polynomial f g z.2, z.1), by fun_prop⟩
  map_zero_left y := by
    change CutOpenCarrier.cylinderMinus F J G ε ε' (polynomial f g y, 0) = _
    rw [CutOpenCarrier.cylinderMinus_zero]
    exact congrArg (CutOpenCarrier.laurentInclusion F J G ε ε')
      (attachment_parameter f g y).symm
  map_one_left y := CutOpenCarrier.cylinderMinus_one F J G ε ε' _

/-- Strict ordinary-sheet descent retains every original polynomial height. -/
def transport : C(SeedOrdinarySheet.Sheet f, Carrier (f ≫ g)) :=
  DoubleCylinder.homotopyDesc
    (SSet.toTop.map (CutOpenCarrier.attaching Ff J Gf εf))
    (SSet.toTop.map (CutOpenCarrier.erase Ff))
    (CutOpenCarrier.first_isPushout Ff J Gf εf)
    (laurent f g) (tipMap f g) (cylinderTrack f g)

theorem transport_cylinder (y : |CutOpenCarrier.parameterSSet Ff|) (s : I) :
    transport f g (SeedOrdinarySheet.cylinder f (y, s)) =
      CutOpenCarrier.cylinderMinus F J G ε ε' (polynomial f g y, s) :=
  DoubleCylinder.homotopyDesc_apply_cylinder _ _ _ _ _ _ s y

theorem transport_laurent (y : |CutOpenCarrier.parameterSSet Gf|) :
    transport f g (SeedOrdinarySheet.laurent f y) = laurent f g y := by
  have ht := ConcreteCategory.congr_hom
    ((CutOpenCarrier.first_isPushout Ff J Gf εf).inr_desc
      (DoubleCylinder.homotopyCylinderMap _ _ _ _ (cylinderTrack f g))
      (DoubleCylinder.homotopyEndMap (laurent f g) (tipMap f g))
      (DoubleCylinder.homotopyDescent_square _ _ _ _ (cylinderTrack f g))) (Sum.inl y)
  exact (congrArg (transport f g)
    (FourEdgeTree.SetDescent.eval_realized_left _ _ y)).trans ht

theorem transport_tip : transport f g (SeedOrdinarySheet.tip f) =
    CutOpenCarrier.tipMinus F J G ε ε' := by
  have ht := ConcreteCategory.congr_hom
    ((CutOpenCarrier.first_isPushout Ff J Gf εf).inr_desc
      (DoubleCylinder.homotopyCylinderMap _ _ _ _ (cylinderTrack f g))
      (DoubleCylinder.homotopyEndMap (laurent f g) (tipMap f g))
      (DoubleCylinder.homotopyDescent_square _ _ _ _ (cylinderTrack f g)))
        (Sum.inr (default : |CutOpenCarrier.tipSSet.{u + 1}|))
  exact (congrArg (transport f g)
    (FourEdgeTree.SetDescent.eval_realized_right _ _ _)).trans ht

attribute [local semireducible] boundedFlatNormalizedPullback

private theorem parameter_native {W : Model.{u + 1}}
    (H : modelOf (boundedFlatPerfectWaldhausen Y) ⟶ W)
    (K : modelOf (boundedFlatPerfectWaldhausen Z) ⟶ W)
    (η : K.functor ≅ (premap g ≫ H).functor)
    (n : ℕ) (y : |CutOpenCarrier.parameterSSet H|)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    pointwisePrespectrumFamily K n (mapParameter (attachPrecompNode H (premap g) K η) y) x =
      pointwisePrespectrumFamily H n y ((boundedFlatPerfectPrespectrumMap g).map n x) := by
  obtain ⟨k, N, z, rfl⟩ := exists_parameter H y
  rw [mapParameter_parameter, pointwisePrespectrumFamily_parameter,
    pointwisePrespectrumFamily_parameter]
  exact prespectrumFamily_attachPrecomp H (premap g) K η N n z x

/-- Every ordinary carrier point is the original precomposed native input. -/
theorem transport_native (n : ℕ) (z : SeedOrdinarySheet.Sheet f)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (transport f g z) x =
      CutOpenCarrier.nativeEval Ff J Gf εf (shifted f) n (SeedOrdinarySheet.inclusion f z)
        ((boundedFlatPerfectPrespectrumMap g).map n x) := by
  rcases Types.eq_or_eq_of_isPushout
    ((CutOpenCarrier.first_isPushout Ff J Gf εf).map (forget TopCat)) z with
      ⟨⟨y, t⟩, hz⟩ | ⟨y | y, hz⟩
  · change DoubleCylinder.realizedCylinder (CutOpenCarrier.attaching Ff J Gf εf)
      (CutOpenCarrier.erase Ff) (y, t) = z at hz
    rw [← hz]
    change CutOpenCarrier.nativeEval F J G ε ε' n
      (transport f g (SeedOrdinarySheet.cylinder f (y, t.down))) x = _
    rw [transport_cylinder]
    change CutOpenCarrier.nativeEval F J G ε ε' n
      (CutOpenCarrier.cylinderMinus F J G ε ε' (polynomial f g y, t.down)) x =
      CutOpenCarrier.nativeEval Ff J Gf εf (shifted f) n
        (CutOpenCarrier.cylinderMinus Ff J Gf εf (shifted f) (y, t.down))
        ((boundedFlatPerfectPrespectrumMap g).map n x)
    rw [CutOpenCarrier.nativeEval_cylinderMinus, CutOpenCarrier.nativeEval_cylinderMinus]
    exact congrArg (fun w ↦ Quillen.ReducedMappingCone.cylinder _ _
      ((InputIdentityPositive.chart X).map n) (t.down, w))
      (parameter_native g Ff F (polynomialAnchor f g) n y x)
  · change DoubleCylinder.realizedEnds (CutOpenCarrier.attaching Ff J Gf εf)
      (CutOpenCarrier.erase Ff) (Sum.inl y) = z at hz
    rw [← hz, ← FourEdgeTree.SetDescent.eval_realized_left]
    change CutOpenCarrier.nativeEval F J G ε ε' n
      (transport f g (SeedOrdinarySheet.laurent f y)) x = _
    rw [transport_laurent]
    change CutOpenCarrier.nativeEval F J G ε ε' n
      (CutOpenCarrier.laurentInclusion F J G ε ε' (mapParameter (laurentNode f g) y)) x =
      CutOpenCarrier.nativeEval Ff J Gf εf (shifted f) n
        (CutOpenCarrier.laurentInclusion Ff J Gf εf (shifted f) y)
        ((boundedFlatPerfectPrespectrumMap g).map n x)
    rw [CutOpenCarrier.nativeEval_laurentInclusion, CutOpenCarrier.nativeEval_laurentInclusion]
    exact congrArg ((mappingConeInclusion (InputIdentityPositive.chart X)).map n)
      (parameter_native g Gf G (laurentAnchor f g) n y x)
  · change DoubleCylinder.realizedEnds (CutOpenCarrier.attaching Ff J Gf εf)
      (CutOpenCarrier.erase Ff) (Sum.inr y) = z at hz
    rw [← hz, ← FourEdgeTree.SetDescent.eval_realized_right]
    have hy : y = (default : |CutOpenCarrier.tipSSet.{u + 1}|) := Subsingleton.elim _ _
    rw [hy]
    change CutOpenCarrier.nativeEval F J G ε ε' n
      (transport f g (SeedOrdinarySheet.tip f)) x = _
    rw [transport_tip]
    change CutOpenCarrier.nativeEval F J G ε ε' n (CutOpenCarrier.tipMinus F J G ε ε') x =
      CutOpenCarrier.nativeEval Ff J Gf εf (shifted f) n
        (CutOpenCarrier.tipMinus Ff J Gf εf (shifted f))
        ((boundedFlatPerfectPrespectrumMap g).map n x)
    rw [CutOpenCarrier.nativeEval_tipMinus, CutOpenCarrier.nativeEval_tipMinus]

/-- The actual second comparison retains both original independent clocks. -/
def surface : C(I × I, Carrier (f ≫ g)) :=
  (transport f g).comp (SeedOrdinarySheet.pastedNull f).toContinuousMap

theorem surface_native (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    (InputIdentityPositive.inclusion X).map n
      (CutOpenCarrier.nativeEval F J G ε ε' n (surface f g (r, s)) x) =
        (InputCompositionNull.secondComparison f g).level n (r, (s, x)) := by
  change (InputIdentityPositive.inclusion X).map n
    (CutOpenCarrier.nativeEval F J G ε ε' n
      (transport f g (SeedOrdinarySheet.pastedNull f (r, s))) x) = _
  rw [transport_native, SeedOrdinarySheet.pastedNull_whole]
  rfl

/-- Typed original old-positive path under the concrete transport. -/
abbrev oldPositive := (SeedOrdinarySheet.oldPositive f).map (transport f g).continuous
/-- Typed original positive movement; its original clock is untouched. -/
abbrev positiveMovement := (SeedOrdinarySheet.positiveMovement f).map (transport f g).continuous
/-- Typed complete polynomial movement under the transport. -/
abbrev polynomialMovement := (SeedOrdinarySheet.polynomialMovement f).map (transport f g).continuous
/-- Typed entire original Laurent movement, not an endpoint replacement. -/
abbrev laurentMovement := (SeedOrdinarySheet.laurentMovement f).map (transport f g).continuous
/-- Typed original new-positive path. -/
abbrev newPositive := (SeedOrdinarySheet.newPositive f).map (transport f g).continuous
/-- The original old cone path retains the actual precomposed cylinder. -/
abbrev oldCone := (SeedOrdinarySheet.oldCone f).map (transport f g).continuous
/-- The original new cone path retains the unchanged precomposed cylinder. -/
abbrev newCone := (SeedOrdinarySheet.newCone f).map (transport f g).continuous

/-- The complete old null retains its original right-associated three-piece word. -/
abbrev oldNull := (oldPositive f g).trans ((positiveMovement f g).trans (oldCone f g))
/-- The complete new null retains its original two-piece word. -/
abbrev newNull := (newPositive f g).trans (newCone f g)

/-- The complete typed comparison retains the literal original source and target associations. -/
def comparison : (oldNull f g).Homotopy ((laurentMovement f g).trans (newNull f g)) :=
  ((SeedOrdinarySheet.pastedNull f).map (transport f g)).cast
    (by
      exact ((SeedOrdinarySheet.oldPositive f).map_trans
        ((SeedOrdinarySheet.positiveMovement f).trans (SeedOrdinarySheet.oldCone f))
        (transport f g).continuous).trans
        (congrArg (fun p ↦ (oldPositive f g).trans p)
          ((SeedOrdinarySheet.positiveMovement f).map_trans (SeedOrdinarySheet.oldCone f)
            (transport f g).continuous)))
    (by
      exact ((SeedOrdinarySheet.laurentMovement f).map_trans
        ((SeedOrdinarySheet.newPositive f).trans (SeedOrdinarySheet.newCone f))
        (transport f g).continuous).trans
        (congrArg (fun p ↦ (laurentMovement f g).trans p)
          ((SeedOrdinarySheet.newPositive f).map_trans (SeedOrdinarySheet.newCone f)
            (transport f g).continuous)))

/-- The typed comparison is precisely the previously exposed entire continuous surface. -/
theorem comparison_apply (r s : I) : comparison f g (r, s) = surface f g (r, s) := rfl

theorem comparison_source (s : I) : surface f g (0, s) = oldNull f g s :=
  (comparison f g).map_zero_left s

theorem comparison_target (s : I) : surface f g (1, s) =
    ((laurentMovement f g).trans (newNull f g)) s :=
  (comparison f g).map_one_left s

/-- The original source cone is the actual unchanged precomposed cylinder. -/
theorem oldCone_apply (s : I) : oldCone f g s =
    CutOpenCarrier.cylinderMinus F J G ε ε' (polynomial f g (movement f 0), s) :=
  transport_cylinder f g _ s

/-- The original target cone retains the same whole precomposed height. -/
theorem newCone_apply (s : I) : newCone f g s =
    CutOpenCarrier.cylinderMinus F J G ε ε' (polynomial f g (movement f 1), s) :=
  transport_cylinder f g _ s
