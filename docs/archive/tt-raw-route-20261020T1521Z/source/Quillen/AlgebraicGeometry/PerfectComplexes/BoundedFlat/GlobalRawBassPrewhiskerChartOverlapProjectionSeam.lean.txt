  (k : laurent ⟶ polynomial)
  (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g

local notation "ε" => InputCompositionPolynomialScalar.anchor f g k

/-- The original Laurent overlap anchor, before source-square transport. -/
def overlapLaurentAnchor : (L).functor ≅ (D ≫ G).functor :=
  (InputCompositionTopPieces.middleAnchor f g).symm ≪≫
    InputCompositionCarrierCharts.chartReferenceIso f g k hk

/-- The full genuine Laurent seam uses the actual normalized source-square isomorphism. -/
def projectionLaurentSeamAnchor :
    (ProjectionReference.reference X (A ≫ E)).functor ≅ (S ≫ G).functor :=
  Functor.associator (A).functor (E).functor (ProjectionReference.source X).functor ≪≫
    sourceSquareAnchor A D S θ L G (overlapLaurentAnchor f g k hk)

/-- Both full attachment paths agree, including the original Laurent normalization anchor. -/
theorem projectionSeam_attachment :
    projectionLaurentSeamAnchor f g c k hk ≪≫ CutOpenPrecomp.anchor S P K G ε =
      ProjectionReference.chartAnchor X k hk (A ≫ E) ≪≫
        isoWhiskerRight (projectionSeamAnchor f g c) (K).functor := by
  have hbase : overlapLaurentAnchor f g k hk ≪≫ isoWhiskerLeft (D).functor ε =
      ProjectionReference.chartAnchor X k hk E ≪≫
        isoWhiskerRight (InputCompositionProjectionAttachment.polynomialAnchor f g) (K).functor :=
          by
    have h := congrArg (fun e ↦ (InputCompositionTopPieces.middleAnchor f g).symm ≪≫ e)
      (InputCompositionProjectionAttachment.attachment_anchor f g k hk)
    simpa only [overlapLaurentAnchor, Iso.trans_assoc, Iso.symm_self_id_assoc] using h.symm
  exact associatedSourceSquare_attachment A D S θ E J (ProjectionReference.source X) K
    (ProjectionReference.chartEndpoint X k hk) P G
    (InputCompositionProjectionAttachment.polynomialAnchor f g)
    (overlapLaurentAnchor f g k hk) ε hbase
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
