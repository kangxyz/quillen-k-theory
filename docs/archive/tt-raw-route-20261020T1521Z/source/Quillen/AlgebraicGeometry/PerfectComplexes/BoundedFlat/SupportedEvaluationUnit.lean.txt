# The original supported forward unit through actual evaluation

The original supported section identity-action and the actual supported geometric unit are
compared through their genuine chosen-zero evaluation isomorphism. Naturality of the raw zero
normalization identifies its forward scalar; realization preserves the whole unit-square
pasting and the independent spectrum structure parameter. No unit filler is assumed.

local instance evaluationRawUnitDownIsEquivalence {A : Scheme.OpenPairs.{u}} :
    (AsSmall.down :
      Waldhausen.SmallModel ((boundedFlatPerfectSupported A.obj.hom).FullSubcategory) ⥤
        (boundedFlatPerfectSupported A.obj.hom).FullSubcategory).IsEquivalence :=
  AsSmall.equiv.symm.isEquivalence_functor

/-- The actual normalized supported raw scalar, including its original universe transport. -/
def evaluationRawUnit {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :
    (evaluationRawMap p).functor ⟶ (evaluationRawMap p).functor :=
  whiskerLeft AsSmall.down (whiskerRight
    (normalizedWeakTransformation (boundedFlatPerfectSupportedPullback_exact p)
      (boundedFlatPerfectSupportedPullback_exact p) (pullbackUnitIso p r).hom) AsSmall.up)

/-- The original raw normalization keeps the same forward target scalar component. -/
theorem evaluationRawUnit_eq {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :
    evaluationRawUnit p r =
      (onFunctor (evaluationRawMap p).functor
        (smallUnit (X := op A.obj.right) A.obj.hom r)).hom := by
  apply NatTrans.ext
  funext K
  change Waldhausen.SmallModel ((boundedFlatPerfectSupported B.obj.hom).FullSubcategory) at K
  apply (AsSmall.down :
    Waldhausen.SmallModel ((boundedFlatPerfectSupported A.obj.hom).FullSubcategory) ⥤
      (boundedFlatPerfectSupported A.obj.hom).FullSubcategory).map_injective
  rw [onFunctor_hom_app, smallUnit, map_lift_hom]
  simp only [evaluationRawUnit, whiskerLeft_app, whiskerRight_app,
    AsSmall.down_map, AsSmall.up_map_down, normalizedWeakTransformation, NatTrans.comp_app,
    pullbackUnitIso, onFunctor_hom_app]
  let z := evaluationRawNormalization p
  have hn := (boundedFlatPerfectSupportedGlobalUnitIso A.obj.hom r).hom.naturality
    (z.hom.app (AsSmall.down.obj K))
  simp only [Functor.id_map, Functor.id_obj] at hn
  change z.inv.app (AsSmall.down.obj K) ≫
      (boundedFlatPerfectSupportedGlobalUnitIso A.obj.hom r).hom.app
        ((boundedFlatPerfectSupportedPullback p).obj (AsSmall.down.obj K)) ≫
          z.hom.app (AsSmall.down.obj K) =
    (boundedFlatPerfectSupportedGlobalUnitIso A.obj.hom r).hom.app
      ((normalizedPullback p).obj (AsSmall.down.obj K))
  rw [← hn, Iso.inv_hom_id_app_assoc]

/-- The entire actual chosen-zero evaluation comparison intertwines the two forward units. -/
theorem evaluation_unit_square {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :
    whiskerRight (onFunctor (supportedPairModelMap p).functor
        (unit (X := op A.obj.right) A.obj.hom r)).hom
        (evaluationModel (X := op A.obj.right) A.obj.hom).functor ≫
      (evaluationComparisonIso p).hom =
    (evaluationComparisonIso p).hom ≫
      whiskerLeft (evaluationModel (X := op B.obj.right) B.obj.hom).functor
        (evaluationRawUnit p r) := by
  rw [evaluationRawUnit_eq]
  exact onFunctor_whisker_square (supportedPairModelMap p).functor
    (evaluationRawMap p).functor (evaluationModel (X := op B.obj.right) B.obj.hom).functor
    (evaluationModel (X := op A.obj.right) A.obj.hom).functor (evaluationComparisonIso p).hom
    (unit (X := op A.obj.right) A.obj.hom r) (smallUnit (X := op A.obj.right) A.obj.hom r)
    (evaluationModel_unit (X := op A.obj.right) A.obj.hom r).whisker

/-- The original supported section identity-action on the actual reindex map. -/
def sectionUnitHomotopy {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :=
  prespectrumIsoHomotopy (supportedPairModelMap p)
    (onFunctor (supportedPairModelMap p).functor (unit (X := op A.obj.right) A.obj.hom r))

/-- The original supported section unit followed by the actual target evaluation. -/
def evaluationOldUnit {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :=
  (sectionUnitHomotopy p r).postcomp
    (prespectrumMap (evaluationModel (X := op A.obj.right) A.obj.hom))

/-- Actual source evaluation followed by the unchanged supported geometric unit. -/
def evaluationNewUnit {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :=
  Homotopy.precomp (prespectrumMap (evaluationModel (X := op B.obj.right) B.obj.hom))
    (pullbackUnitHomotopy p r)

/-- The original supported unit square realizes to its complete two-route pasting. -/
def evaluationUnitPasting {A B : Scheme.OpenPairs.{u}} (p : A ⟶ B)
    (r : (Γ(A.obj.right, ⊤))ˣ) :
    ((evaluationOldUnit p r).trans (evaluationComparison p)).RelHomotopy
      ((evaluationComparison p).trans (evaluationNewUnit p r)) := by
  let h := boundedFlatPerfectSupportedPullback_exact p
  let J := prespectrumTransformationIntertwiningHomotopy
    (supportedPairModelMap p) (evaluationRawMap p)
    (evaluationModel (X := op B.obj.right) B.obj.hom)
    (evaluationModel (X := op A.obj.right) A.obj.hom)
    (onFunctor (supportedPairModelMap p).functor
      (unit (X := op A.obj.right) A.obj.hom r)).hom
    (evaluationRawUnit p r) (evaluationComparisonIso p).hom
    (fun _ ↦ (supportedModel (X := op A.obj.right) A.obj.hom).waldhausen.weq_iso _ inferInstance)
    (fun K ↦ normalizedWeakTransformation_weak h h (pullbackUnitIso p r).hom
      (fun _ ↦ (boundedFlatPerfectSupportedWaldhausen A.obj.hom).weq_iso _ inferInstance)
      (AsSmall.down.obj K))
    (fun K ↦ (boundedFlatPerfectSupportedWaldhausen A.obj.hom).smallModel.weq_iso _
      (inferInstanceAs (IsIso ((evaluationComparisonIso p).app K).hom)))
    (evaluation_unit_square p r)
  refine J.ofLevels _ _ ?_ ?_
  · intro n
    apply Homotopy.trans_level_congr <;> rfl
  · intro n
    apply Homotopy.trans_level_congr <;> rfl
