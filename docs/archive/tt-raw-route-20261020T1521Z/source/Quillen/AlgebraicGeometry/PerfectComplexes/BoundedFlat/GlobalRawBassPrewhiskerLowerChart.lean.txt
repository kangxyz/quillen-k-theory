module
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassPrewhiskerLowerLaurentNative
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.NativeSquareFiveSimplex
/-! All-point recovery of the original source-map first half with all three clocks retained. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Quillen.Waldhausen.IteratedS Quillen.SimplexParameters
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open scoped Simplicial Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
open CommonReferenceNerve
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

variable (k : laurent ⟶ polynomial)
local notation "S" => NativeSquareFiveSimplex.source (laurent.map c) (k.app W)
local notation "M" => NativeSquareFiveSimplex.nodes (laurent.map c) (k.app W)
  (k.app Z) (polynomial.map c) (k.naturality c)

/-- The genuine polynomial-source lower half retains the original chart-square reference. -/
abbrev lowerChartReference := S ≫ lowerLaurentBinary f g

/-- The original three-node composition acts on the original five-node chart square. -/
def lowerChart : C(I × I, |CutOpenCarrier.parameterSSet (lowerChartReference f g c k)|) :=
  (FinitePrecomp.parameter S M (lowerLaurentBinary f g)).comp
    ⟨fun z ↦ (lowerLaurentPath f g z.1, NativeSquareFiveSimplex.parameter z.2), by fun_prop⟩

/-- Every genuine polynomial input retains both original independent lower-half clocks. -/
theorem lowerChart_native (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj W))).space n) :
    pointwisePrespectrumFamily (lowerChartReference f g c k) n (lowerChart f g c k (r, s)) x =
      (InputCompositionCharts.laurentComposition f g).level n
        (r, (boundedFlatPerfectPrespectrumSquareHomotopy (laurent.map c) (k.app W)
          (k.app Z) (polynomial.map c) (k.naturality c)).level n (s, x)) := by
  change pointwisePrespectrumFamily _ n
    (FinitePrecomp.parameter S M (lowerLaurentBinary f g)
      (lowerLaurentPath f g r, NativeSquareFiveSimplex.parameter s)) x = _
  rw [FinitePrecomp.parameter_native, lowerLaurentPath_native, prespectrumFamily_eq]
  exact congrArg (fun y ↦ (InputCompositionCharts.laurentComposition f g).level n (r, y))
    (NativeSquareFiveSimplex.native (laurent.map c) (k.app W) (k.app Z)
      (polynomial.map c) (k.naturality c) n s x)
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
