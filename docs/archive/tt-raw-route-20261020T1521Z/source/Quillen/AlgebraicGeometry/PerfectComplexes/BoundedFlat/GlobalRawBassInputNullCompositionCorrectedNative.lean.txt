local notation "F" => InputIdentitySeedCarrier.polyRef (f ≫ g)
local notation "G" => InputIdentitySeedCarrier.laurentRef (f ≫ g)
local notation "J" => InputIdentitySeedCarrier.chart (X := X)

/-- The first original chart half is exactly the first native associator piece. -/
theorem finalChartFirst_native (n : ℕ) (s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (finalChartFirst f g s) x =
      (ConeCompositionPieces.first h a d (plusSquareHomotopy g)).level n
        (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x) := by
  change CutOpenCarrier.nativeEval F J G ε ε' n
    (nullMovingFirst f g (1, lower s)) x = _
  rw [nullMovingFirst_native]
  have ht := (InputCompositionPositive.comparisonSquare f g).at_one n (lower s)
    ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
  change (InputCompositionCharts.plusChart f g).level n (1, (lower s, _)) =
    (InputCompositionPositive.compositeComparison f g).level n (lower s, _) at ht
  rw [ht]
  unfold InputCompositionPositive.compositeComparison coneSquarePastedHomotopy
  rw [Homotopy.cast_level, Homotopy.trans_lower_cell]
  rfl

/-- The second original chart half is exactly the second native associator piece. -/
theorem finalChartSecond_native (n : ℕ) (s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (finalChartSecond f g s) x =
      (ConeCompositionPieces.second h a c d (plusSquareHomotopy f)).level n
        (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x) := by
  change CutOpenCarrier.nativeEval F J G ε ε' n
    (nullMovingFirst f g (1, upper s)) x = _
  rw [nullMovingFirst_native]
  have ht := (InputCompositionPositive.comparisonSquare f g).at_one n (upper s)
    ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
  change (InputCompositionCharts.plusChart f g).level n (1, (upper s, _)) =
    (InputCompositionPositive.compositeComparison f g).level n (upper s, _) at ht
  rw [ht]
  unfold InputCompositionPositive.compositeComparison coneSquarePastedHomotopy
  rw [Homotopy.cast_level, Homotopy.trans_upper_cell]
  rfl

/-- The last actual cylinder is the unchanged third native associator piece. -/
theorem finalCylinder_native (n : ℕ) (s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (finalCylinder f g s) x =
      (ConeCompositionPieces.third h a c).level n
        (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x) := by
  change CutOpenCarrier.nativeEval F J G ε ε' n (nullMovingSecond f g (1, s)) x = _
  rw [nullMovingSecond_native, Homotopy.at_one]
  rfl

/-- The exact original correction, without replacing its associator interior or either clock. -/
def nativeCorrection := coneSquareCompositionNullComparison
  (InputIdentityPositive.chart Z) (InputIdentityPositive.chart Y) h a b c d
  (plusSquareHomotopy g) (plusSquareHomotopy f)

/-- Every value of the scalar correction recovers the already chosen native correction. -/
theorem finalCorrection_native (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (finalCorrection f g (r, s)) x =
      (nativeCorrection f g).level n
        (r, (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)) := by
  unfold nativeCorrection
  rw [ConeCompositionPieces.correction_apply]
  change CutOpenCarrier.nativeEval F J G ε ε' n
    (Path.Homotopy.transAssoc (finalChartFirst f g) (finalChartSecond f g)
      (finalCylinder f g) (r, s)) x = _
  exact Path.Evaluation.transAssoc (fun z ↦ CutOpenCarrier.nativeEval F J G ε ε' n z x)
    _ _ _ _ _ _ (fun t ↦ finalChartFirst_native f g n t x)
    (fun t ↦ finalChartSecond_native f g n t x)
    (fun t ↦ finalCylinder_native f g n t x) r s


private theorem comparisonTrans_native {V : Type*} [TopologicalSpace V]
    {v w : V} {p q t : Path v w} {E T : LoopPrespectrum.{u}} {i j : E ⟶ T}
    {H K L : Homotopy i j} (n : ℕ) (x : E.space n) (ev : V → T.space n)
    (P : p.Homotopy q) (Q : q.Homotopy t)
    (P' : H.RelHomotopy K) (Q' : K.RelHomotopy L)
    (hp : ∀ r s, ev (P (r, s)) = P'.level n (r, (s, x)))
    (hq : ∀ r s, ev (Q (r, s)) = Q'.level n (r, (s, x))) (r s : unitInterval) :
    ev (P.trans Q (r, s)) = (P'.trans Q').level n (r, (s, x)) := by
  rw [Homotopy.RelHomotopy.trans_apply_path]
  exact Path.Evaluation.comparison_trans ev P Q (P'.path n x) (Q'.path n x) hp hq r s

private theorem comparisonHcomp_native {V : Type*} [TopologicalSpace V]
    {v w z : V} {p q : Path v w} {t u : Path w z}
    {E T : LoopPrespectrum.{u}} {i j k : E ⟶ T}
    {H H' : Homotopy i j} {K K' : Homotopy j k}
    (n : ℕ) (x : E.space n) (ev : V → T.space n)
    (P : p.Homotopy q) (Q : t.Homotopy u)
    (P' : H.RelHomotopy H') (Q' : K.RelHomotopy K')
    (hp : ∀ r s, ev (P (r, s)) = P'.level n (r, (s, x)))
    (hq : ∀ r s, ev (Q (r, s)) = Q'.level n (r, (s, x))) (r s : unitInterval) :
    ev (P.hcomp Q (r, s)) = (P'.hcomp Q').level n (r, (s, x)) := by
  rw [Homotopy.RelHomotopy.hcomp_apply_path]
  exact Path.Evaluation.comparison_hcomp ev P Q (P'.path n x) (Q'.path n x) hp hq r s

/-- The native three-side boundary uses the actual bottom track and terminal zero pause. -/
def nativeBoundary := (nativeMovingSquare f g).toBoundary.trans
  (((Homotopy.RelHomotopy.refl
    (Homotopy.precomp (InputIdentityPositive.chart Z)
      ((InputCompositionCharts.laurentComposition f g).postcomp (mappingConeInclusion h)))).hcomp
        (nativeCorrection f g)).hcomp
    (Homotopy.RelHomotopy.refl (Quillen.LoopPrespectrum.Homotopy.refl
      (Quillen.LoopPrespectrum.zeroHom
        (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj Z)))
        (mappingCone h))).symm))

/-- The whole corrected scalar boundary recovers the prescribed native composition boundary. -/
theorem correctedBoundary_native (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (correctedBoundary f g (r, s)) x =
      (nativeBoundary f g).level n
        (r, (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)) := by
  unfold correctedBoundary nativeBoundary
  refine comparisonTrans_native n
    ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
    (fun z ↦ CutOpenCarrier.nativeEval F J G ε ε' n z x) _ _ _ _ ?_ ?_ r s
  · intro t v
    exact movingBoundary_native f g n t v x
  · intro t v
    refine comparisonHcomp_native n
      ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
      (fun z ↦ CutOpenCarrier.nativeEval F J G ε ε' n z x) _ _ _ _ ?_ ?_ t v
    · intro t' v'
      refine comparisonHcomp_native n
        ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
        (fun z ↦ CutOpenCarrier.nativeEval F J G ε ε' n z x) _ _ _ _ ?_ ?_ t' v'
      · intro t'' v''
        change CutOpenCarrier.nativeEval F J G ε ε' n (nullMovingFirst f g (v'', 0)) x = _
        rw [nullMovingFirst_native]
        have hc := (InputCompositionPositive.comparisonSquare f g).left n v''
          ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
        exact congrArg ((mappingConeInclusion h).map n) hc
      · intro t'' v''
        exact finalCorrection_native f g n t'' v'' x
    · intro t' v'
      change CutOpenCarrier.nativeEval F J G ε ε' n
        (InputIdentitySeedCone.tip (f ≫ g) false) x = _
      exact CutOpenCarrier.nativeEval_tipMinus F J G ε ε' n x

/-- The actual quotient descent recovers the entire already chosen corrected native square. -/
theorem correctedSquare_native (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n ((correctedSquare f g).parameter (r, s)) x =
      (InputCompositionPositive.nullSquare f g).level n
        (r, (s, (boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)) := by
  exact Quillen.CarrierPathSquare.boundarySquare_native (correctedBoundary f g)
    (nativeBoundary f g) n ((boundedFlatPerfectPrespectrumMap (projection.app Z)).map n x)
    (fun z ↦ CutOpenCarrier.nativeEval F J G ε ε' n z x)
    (fun t v ↦ correctedBoundary_native f g n t v x) r s

/-- The explicit scalar source rail recovers the whole original positive source rail. -/
theorem sourceRail_native (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    CutOpenCarrier.nativeEval F J G ε ε' n (sourceRail f g (r, s)) x =
      InputCompositionPositive.sourceRail f g n (r, (s, x)) := by
  rcases cover s with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [sourceRail_lower, sourceFirst_native]
    change _ = (InputCompositionPositive.coneHomotopy f g).level n
      (r, (InputIdentityPositive.sourceNull Z).level n (lower t, x))
    unfold InputIdentityPositive.sourceNull
    rw [Homotopy.trans_lower_cell]
    rfl
  · rw [sourceRail_upper, correctedSquare_native]
    change _ = (InputCompositionPositive.coneHomotopy f g).level n
      (r, (InputIdentityPositive.sourceNull Z).level n (upper t, x))
    unfold InputIdentityPositive.sourceNull
    rw [Homotopy.trans_upper_cell]
    rfl

/-- The exact same scalar surface recovers the selected original whole-cone source rail. -/
theorem sourceRail_original (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    (InputIdentityPositive.inclusion X).map n
      (CutOpenCarrier.nativeEval F J G ε ε' n (sourceRail f g (r, s)) x) =
        (InputComposition.coneHomotopy f g).level n
          (r, (GlobalRawBass.sourceNull Z).level n (s, x)) := by
  rw [sourceRail_native]
  exact InputCompositionPositive.sourceRail_factor f g n r s x
