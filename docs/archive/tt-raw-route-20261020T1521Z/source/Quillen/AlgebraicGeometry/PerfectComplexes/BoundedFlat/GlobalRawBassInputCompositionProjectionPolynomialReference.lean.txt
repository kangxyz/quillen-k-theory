variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "D" => InputCompositionCarrierCharts.projectionMap (Z := Z)

/-- Polynomial Anchor for the exact original reference and parameter data. -/
def polynomialAnchor : (T).functor ≅ (D ≫ P).functor :=
  (InputCompositionSpanScalar.polynomialAnchor f g).symm ≪≫
    InputCompositionTopChartRows.movingBaseAnchor f g

/-- Polynomial Node for the exact original reference and parameter data. -/
def polynomialNode (N : Node P) : Node T :=
  reanchor (polynomialAnchor f g) (precompNode D N)
