/-!
# The positive middle seam between the transported chosen nulls

The exact Laurent reference identification transports the original three-node K/D seam
to an equality of paths in the composite cut-open carrier. Both binary clocks remain
unchanged; no native evaluation injectivity is assumed.
-/

open CommonReferenceNerve
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- Exact reference matching applies to every point of the first-arrow Laurent nerve. -/
theorem pre_parameter (y : |CutOpenCarrier.parameterSSet (InputIdentitySeedCarrier.laurentRef f)|) :
    mapParameter (InputNullCompositionPrecomp.laurentNode f g) y =
      mapParameter (InputCompositionTerminalAttach.preNode f g) y := by
  obtain ⟨k, N, z, rfl⟩ := exists_parameter (InputIdentitySeedCarrier.laurentRef f) y
  rw [mapParameter_parameter, mapParameter_parameter]
  exact congrArg (fun ns ↦ parameter (InputIdentitySeedCarrier.laurentRef (f ≫ g)) ns z)
    (funext fun i ↦ InputNullCompositionPrecomp.laurent_node f g (N i))

/-- The transported positive paths meet with the original full binary time, not a new speed. -/
theorem newPositive_oldPositive (s : I) :
    SeedSurfaceTransport.newPositive f g s = InputNullCompositionPrecomp.oldPositive f g s := by
  change SeedSurfaceTransport.transport f g
    (SeedOrdinarySheet.laurent g (parameter (InputDescSimplex.reference g)
      (InputDescSimplex.nodes g) (ProjectionRectangleParameters.K BassH0Diagram.plusFace s))) =
    InputNullCompositionPrecomp.transport f g
      (SeedOrdinarySheet.laurent f (parameter (InputDescSimplex.reference f)
        (InputDescSimplex.nodes f) (ProjectionRectangleParameters.D BassH0Diagram.plusFace s)))
  rw [SeedSurfaceTransport.transport_laurent, InputNullCompositionPrecomp.transport_laurent]
  change CutOpenCarrier.laurentInclusion _ _ _ _ _
    (mapParameter (InputCompositionTerminalAttach.postNode f g)
      (parameter _ _ (ProjectionRectangleParameters.K BassH0Diagram.plusFace s))) =
    CutOpenCarrier.laurentInclusion _ _ _ _ _
      (mapParameter (InputNullCompositionPrecomp.laurentNode f g)
        (parameter _ _ (ProjectionRectangleParameters.D BassH0Diagram.plusFace s)))
  rw [pre_parameter]
  exact congrArg (CutOpenCarrier.laurentInclusion _ _ _ _ _) (positive_parameter f g s)

end Quillen.PerfectComplexes.GlobalRawBass.SeedPositiveSeam
