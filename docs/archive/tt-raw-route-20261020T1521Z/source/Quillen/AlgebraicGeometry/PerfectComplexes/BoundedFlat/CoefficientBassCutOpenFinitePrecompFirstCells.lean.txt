/-- The original realized left summand is the exact displayed endpoint map. -/
theorem realized_left {X Y Z : SSet.{u}} (f : X ⟶ Y) (g : X ⟶ Z)
    (y : |Y|) :
    SSet.toTop.map ((coprod.inl : Y ⟶ Y ⨿ Z) ≫
      DoubleCylinder.simplicialEnds f g) y =
        DoubleCylinder.realizedEnds f g (Sum.inl y) := by
  have h : SSet.toTop.map ((coprod.inl : Y ⟶ Y ⨿ Z) ≫
      DoubleCylinder.simplicialEnds f g) =
      (TopCat.binaryCofan |Y| |Z|).inl ≫ DoubleCylinder.realizedEnds f g := by
    simpa only [Functor.map_comp, DoubleCylinder.realizedEnds, Category.assoc,
      Iso.hom_inv_id_assoc] using
      congrArg (fun k ↦ k ≫ (realizationBinarySumIso Y Z).inv ≫
        SSet.toTop.map (DoubleCylinder.simplicialEnds f g))
        (realizationBinarySumIso_inl Y Z)
  exact ConcreteCategory.congr_hom h y

/-- The original realized right summand is the exact displayed endpoint map. -/
theorem realized_right {X Y Z : SSet.{u}} (f : X ⟶ Y) (g : X ⟶ Z)
    (z : |Z|) :
    SSet.toTop.map ((coprod.inr : Z ⟶ Y ⨿ Z) ≫
      DoubleCylinder.simplicialEnds f g) z =
        DoubleCylinder.realizedEnds f g (Sum.inr z) := by
  have h : SSet.toTop.map ((coprod.inr : Z ⟶ Y ⨿ Z) ≫
      DoubleCylinder.simplicialEnds f g) =
      (TopCat.binaryCofan |Y| |Z|).inr ≫ DoubleCylinder.realizedEnds f g := by
    simpa only [Functor.map_comp, DoubleCylinder.realizedEnds, Category.assoc,
      Iso.hom_inv_id_assoc] using
      congrArg (fun k ↦ k ≫ (realizationBinarySumIso Y Z).inv ≫
        SSet.toTop.map (DoubleCylinder.simplicialEnds f g))
        (realizationBinarySumIso_inr Y Z)
  exact ConcreteCategory.congr_hom h z

variable {U V W T : Model.{u}} (A : U ⟶ V)
variable {q : ℕ} (M : Fin (q + 1) → Node A)
variable (F : V ⟶ W) (J : W ⟶ T) (G : V ⟶ T)
variable (ε₀ ε₁ : G.functor ≅ (F ≫ J).functor)

/-- The whole Laurent reference summand is retained after first-sheet descent. -/
theorem first_left (y : |parameterSSet G|) :
    first A M F J G ε₀ ε₁ (SSet.toTop.map (left₀ F J G ε₀) y) =
      laurent A M F J G ε₀ ε₁ y := by
  rw [show SSet.toTop.map (left₀ F J G ε₀) y =
    DoubleCylinder.realizedEnds (attaching F J G ε₀) (erase F) (Sum.inl y) from
      realized_left _ _ y, first_ends]
  rfl

/-- The separately retained negative tip is constant in the finite source parameter. -/
theorem first_right (y : |tipSSet.{u}|) :
    first A M F J G ε₀ ε₁ (SSet.toTop.map (right₀ F J G ε₀) y) =
      ContinuousMap.const _ (tipMinus (A ≫ F) J (A ≫ G)
        (CutOpenPrecomp.anchor A F J G ε₀) (CutOpenPrecomp.anchor A F J G ε₁)) := by
  rw [show SSet.toTop.map (right₀ F J G ε₀) y =
    DoubleCylinder.realizedEnds (attaching F J G ε₀) (erase F) (Sum.inr y) from
      realized_right _ _ y, first_ends]
  rfl

end Quillen.Waldhausen.IteratedS.CommonReferenceNerve.FinitePrecomp
