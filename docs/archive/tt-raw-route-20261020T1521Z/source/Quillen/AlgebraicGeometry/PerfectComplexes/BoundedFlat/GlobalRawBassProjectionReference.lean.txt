/-!
# Whole-reference nodes of the original global Bass projection

The unchanged `InputIdentityProjection` five-node path acts on a whole Model reference.
Both chart maps accept arbitrary polynomial-reference nodes. Their full endpoint attachments
retain the actual positive and inverse global projection anchors, before native evaluation.
These are the actual input nodes for a global four-edge scalar presentation; no raw cone
family or higher filler is assumed.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory CategoryTheory.Functor AlgebraicGeometry

namespace Quillen.PerfectComplexes.GlobalRawBass.ProjectionReference
open CommonReferenceNerve CommonReferenceProduct
open InputCompositionTerminalAttach
attribute [local irreducible] boundedFlatNormalizedPullback
variable (X : Scheme.{u})

/-- The actual normalized pullback along the original global polynomial projection. -/
abbrev projectionMap := arrow (projection.app X)

/-- The source is exactly the shared source of the original global five-node projection path. -/
abbrev source := InputIdentityProjection.source X

variable (k : laurent ⟶ polynomial)
  (hk : ∀ Y, k.app Y ≫ projection.app Y = laurentProjection.app Y)

/-- The full source change uses the actual chart equation, with the original positive source. -/
def chartSourceIso : (source X).functor ≅
    (BinaryNormalizationSimplex.sourceMap (k.app X) (projection.app X)).functor :=
  eqToIso (congrArg (fun F ↦ AsSmall.down ⋙ F ⋙ AsSmall.up)
    (congrArg boundedFlatNormalizedPullback ((plus_projection X).trans (hk X).symm)))

/-- The entire original chart endpoint retains both geometric and normalization components. -/
def chartEndpoint : (source X).functor ≅
    (projectionMap X ≫ arrow (k.app X)).functor :=
  chartSourceIso X k hk ≪≫ BinaryNormalizationSimplex.total (k.app X) (projection.app X)

/-- The positive endpoint is the complete old node, including its original reference. -/
theorem plus_endpoint : InputIdentityProjection.nodes X 2 =
    (⟨projectionMap X ≫ arrow (plus.app X), chartEndpoint X plus plus_projection⟩ :
      Node (source X)) := by
  apply Sigma.ext (Model.Hom.ext rfl)
  apply heq_of_eq
  rfl

/-- The inverse endpoint retains its separate original source transport. -/
theorem minus_endpoint : InputIdentityProjection.nodes X 4 =
    (⟨projectionMap X ≫ arrow (minus.app X), chartEndpoint X minus minus_projection⟩ :
      Node (source X)) := by
  apply Sigma.ext (Model.Hom.ext rfl)
  apply heq_of_eq
  rfl

variable {V : Model.{u + 1}}
  (E : V ⟶ modelOf (boundedFlatPerfectWaldhausen X))

/-- The full Laurent reference is the literal composite with the unchanged global path source. -/
abbrev reference := E ≫ source X

/-- The polynomial reference keeps the original global projection as its second factor. -/
abbrev polynomialReference := E ≫ projectionMap X

/-- Both factors retain their entire dependent-node isomorphisms. -/
def nodes (N : Node E) (i : Fin 5) : Node (reference X E) :=
  node (reference X E) E (source X) (Iso.refl _) N (InputIdentityProjection.nodes X i)

/-- Every actual source node is postcomposed by the original polynomial projection. -/
def polynomialNode (N : Node E) : Node (polynomialReference X E) :=
  postcompNode (projectionMap X) N

/-- The entire chart anchor is whiskered before any polynomial node is supplied. -/
def chartAnchor : (reference X E).functor ≅
    (polynomialReference X E ≫ arrow (k.app X)).functor :=
  isoWhiskerLeft E.functor (chartEndpoint X k hk)

/-- The chart attachment accepts arbitrary genuine polynomial-reference nodes. -/
def chartNode (M : Node (polynomialReference X E)) : Node (reference X E) :=
  postNode (reference X E) (arrow (k.app X)) (chartAnchor X k hk E) M

private theorem endpoint_product (N : Node E) :
    chartNode X k hk E (polynomialNode X E N) =
      node (reference X E) E (source X) (Iso.refl _) N
        ⟨projectionMap X ≫ arrow (k.app X), chartEndpoint X k hk⟩ := by
  apply Sigma.ext
  · exact Category.assoc N.1 (projectionMap X) (arrow (k.app X))
  apply heq_of_eq
  apply Iso.ext
  ext A
  simp only [chartNode, polynomialNode, chartAnchor, CommonReferenceProduct.postNode,
    reanchor, postcompNode, node,
    Iso.refl_trans, BassReferenceMerge.composition_pair_iso,
    Iso.trans_hom, NatTrans.comp_app, isoWhiskerLeft_hom, isoWhiskerRight_hom,
    whiskerLeft_app, whiskerRight_app]
  exact ((chartEndpoint X k hk).hom.naturality (N.2.hom.app A)).symm

/-- The complete positive polynomial attachment is the original endpoint of the five-node path. -/
theorem plus_attachment (N : Node E) :
    chartNode X plus plus_projection E (polynomialNode X E N) = nodes X E N 2 := by
  rw [nodes, plus_endpoint]
  exact endpoint_product X plus plus_projection E N

/-- The complete inverse attachment keeps its distinct original endpoint and full anchor. -/
theorem minus_attachment (N : Node E) :
    chartNode X minus minus_projection E (polynomialNode X E N) = nodes X E N 4 := by
  rw [nodes, minus_endpoint]
  exact endpoint_product X minus minus_projection E N

end Quillen.PerfectComplexes.GlobalRawBass.ProjectionReference
