# Actual global polynomial composition nodes before the target Laurent attachment

The original eighteen-node polynomial rectangle is reanchored into the genuine polynomial
summands of the global `InputIdentityProjection` four-edge tree. Both chart attachments
recover the complete original moving nodes and all characteristic parameters, using the
existing normalized associativity identity. No native injectivity or scalar filler is assumed.
-/

variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
local notation "E" => baseSource f g
local notation "Lref" => ProjectionReference.reference X E

/-- The actual polynomial rectangle source changes to the target projection reference. -/
def polynomialAnchor : (polynomialSource f g).functor ≅
    (ProjectionReference.polynomialReference X E).functor :=
  BinaryNormalizationSimplex.total (projection.app X) (f ≫ g)

/-- Every complete polynomial node enters the actual target tree before chart postcomposition. -/
def polynomialNode (N : Node (polynomialSource f g)) :
    Node (ProjectionReference.polynomialReference X E) :=
  reanchor (polynomialAnchor f g).symm N

variable (k : laurent ⟶ polynomial)
  (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)

/-- The new whole-reference endpoint is the original global normalization endpoint exactly. -/
theorem projectionEndpoint_eq : ProjectionReference.chartEndpoint X k hk =
    projectionEndpointIso (X := X) k hk :=
  BassMJoin.small_binary_transport (plus.app X ≫ projection.app X)
    (k.app X) (projection.app X) ((plus_projection X).trans (hk X).symm)

/-- The complete original chart anchor factors through the actual polynomial-reference change. -/
theorem moving_anchor :
    (middleAnchor f g ≪≫ ProjectionReference.chartAnchor X k hk E) ≪≫
      isoWhiskerRight (polynomialAnchor f g).symm (chartMap (X := X) k).functor =
        movingAnchor f g k hk := by
  have h := internal_reference f g k hk
  change movingAnchor f g k hk ≪≫
      isoWhiskerRight (polynomialAnchor f g) (chartMap (X := X) k).functor =
    middleAnchor f g ≪≫ isoWhiskerLeft (E).functor
      (projectionEndpointIso (X := X) k hk) at h
  rw [← projectionEndpoint_eq k hk] at h
  change (middleAnchor f g ≪≫ isoWhiskerLeft (E).functor
      (ProjectionReference.chartEndpoint X k hk)) ≪≫ _ = _
  rw [← h, Iso.trans_assoc, ← isoWhiskerRight_trans]
  simp only [Iso.self_symm_id, isoWhiskerRight_refl]
  apply Iso.ext
  ext A
  exact Category.comp_id _

/-- All genuine polynomial nodes, including their anchors, have the prescribed chart attachment. -/
theorem moving_node_attachment (N : Node (polynomialSource f g)) :
    reanchor (middleAnchor f g)
        (ProjectionReference.chartNode X k hk E (polynomialNode f g N)) =
      (⟨N.1 ≫ chartMap (X := X) k,
        movingAnchor f g k hk ≪≫ isoWhiskerRight N.2 (chartMap (X := X) k).functor⟩ :
          InputCompositionCarrierCharts.Node f g) := by
  apply Sigma.ext
  · exact Model.Hom.ext rfl
  apply heq_of_eq
  change middleAnchor f g ≪≫ (ProjectionReference.chartAnchor X k hk E ≪≫
    isoWhiskerRight ((polynomialAnchor f g).symm ≪≫ N.2)
      (chartMap (X := X) k).functor) = _
  rw [isoWhiskerRight_trans]
  simpa only [Iso.trans_assoc] using congrArg
    (fun e ↦ e ≪≫ isoWhiskerRight N.2 (chartMap (X := X) k).functor)
    (moving_anchor f g k hk)

/-- Every one of the eighteen original moving nodes is retained in its complete old reference. -/
theorem moving_nodes (i : Fin 18) :
    reanchor (middleAnchor f g)
        (ProjectionReference.chartNode X k hk E (polynomialNode f g (polynomialNodes f g i))) =
      movingNodes f g k hk i :=
  moving_node_attachment f g k hk (polynomialNodes f g i)

/-- The entire old characteristic simplex is recovered, before evaluating a native point. -/
theorem moving_parameter (z : StdSimplex ℝ (Fin 18)) :
    mapParameter (reanchor (middleAnchor f g))
        (SSet.toTop.map (ProjectionReference.FourEdge.chartMap X E k hk)
          (parameter (ProjectionReference.polynomialReference X E)
            (fun i ↦ polynomialNode f g (polynomialNodes f g i)) z)) =
      parameter (InputCompositionCarrierCharts.source f g) (movingNodes f g k hk) z := by
  change mapParameter (reanchor (middleAnchor f g))
    (mapParameter (ProjectionReference.chartNode X k hk E) _) = _
  rw [mapParameter_parameter]
  have h := mapParameter_parameter (F := Lref)
    (G := InputCompositionCarrierCharts.source f g)
    (reanchor (middleAnchor f g))
    (fun i : Fin 18 ↦ ProjectionReference.chartNode X k hk E
      (polynomialNode f g (polynomialNodes f g i))) z
  exact h.trans (congrArg
    (fun N ↦ parameter (InputCompositionCarrierCharts.source f g) N z)
    (funext (moving_nodes f g k hk)))
