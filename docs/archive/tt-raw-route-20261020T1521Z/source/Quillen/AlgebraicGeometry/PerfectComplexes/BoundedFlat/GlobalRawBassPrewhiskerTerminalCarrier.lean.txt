variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
  (k : laurent ⟶ polynomial) (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "S" => NativeSquareFiveSimplex.source (polynomial.map c) (projection.app W)
local notation "M" => NativeSquareFiveSimplex.nodes (polynomial.map c) (projection.app W)
  (projection.app Z) c (projection.naturality c)
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "G" => InputCompositionPolynomialScalar.laurentReference f g k
local notation "K" => InputCompositionPolynomialScalar.chart (X := X) k
local notation "ε" => InputCompositionPolynomialScalar.anchor f g k
local notation "J" => ProjectionReference.projectionMap X
local notation "L'" => ProjectionReference.reference X (A ≫ E)
local notation "β'" => ProjectionReference.FourEdge.beta X (A ≫ E)

private theorem terminal_carrier
    (v0 : parameterSSet (E ≫ J) ⟶ FourEdgeTree.simplicial E J)
    (h0 : v0 ≫ ProjectionReference.FourEdge.beta X E =
      ProjectionReference.FourEdge.chartMap X E k hk)
    (v1 : parameterSSet ((A ≫ E) ≫ J) ⟶ FourEdgeTree.simplicial (A ≫ E) J)
    (h1 : v1 ≫ β' = ProjectionReference.FourEdge.chartMap X (A ≫ E) k hk)
    (hc : ∀ y s, preCylinderMap f g c v1
      (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4), s) =
      ProjectionReference.FourEdge.Precomp.cone X A E
        (InputCompositionProjectionAttachment.cylinderMap f g v0 (y, s)))
    (y : InputCompositionPolynomialScalar.Carrier f g k) :
    preConeMap f g c k hk v1 h1
      (FinitePrecomp.precompFinite S M P K G ε ε (y, Convexity.StdSimplex.single 4)) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (InputCompositionProjectionAttachment.coneMap f g k hk v0 h0 y) := by
  rcases exists_native_cell P K G ε ε y with
    ⟨z, rfl⟩ | ⟨z, t, rfl⟩ | ⟨z, t, rfl⟩ | rfl | rfl
  · simp only [FinitePrecomp.precompFinite_laurent, preConeMap,
      InputCompositionProjectionAttachment.coneMap, CutOpenFold.map_laurent]
    change FourEdgeTree.laurentInclusion (A ≫ E) J L' β'
      (mapParameter (reanchor (projectionLaurentSeamAnchor f g c k hk))
        (FinitePrecomp.parameter S M G (z, Convexity.StdSimplex.single 4))) =
      ProjectionReference.FourEdge.Precomp.cone X A E
        (FourEdgeTree.laurentInclusion E J (ProjectionReference.reference X E)
          (ProjectionReference.FourEdge.beta X E)
          (mapParameter (InputCompositionProjectionAttachment.laurentNode f g k hk) z))
    rw [ProjectionReference.FourEdge.Precomp.cone_laurent]
    exact congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L' β')
      (terminal_laurent_parameter f g c k hk z)
  · simpa only [FinitePrecomp.precompFinite_cylinderMinus, preConeMap,
      InputCompositionProjectionAttachment.coneMap, CutOpenFold.map_cylinder] using hc z t
  · simpa only [FinitePrecomp.precompFinite_cylinderPlus, preConeMap,
      InputCompositionProjectionAttachment.coneMap, CutOpenFold.map_cylinderPlus] using hc z t
  · simp only [FinitePrecomp.precompFinite_tipMinus, preConeMap,
      InputCompositionProjectionAttachment.coneMap, CutOpenFold.map_tipMinus,
      ProjectionReference.FourEdge.Precomp.cone_tip]
  · simp only [FinitePrecomp.precompFinite_tipPlus, preConeMap,
      InputCompositionProjectionAttachment.coneMap, CutOpenFold.map_tipPlus,
      ProjectionReference.FourEdge.Precomp.cone_tip]

/-- The whole positive source-square carrier meets the actual fixed-source overlap map. -/
theorem terminal_plus_carrier (y : InputCompositionPolynomialScalar.Carrier f g plus) :
    prePlusConeMap f g c
      (FinitePrecomp.precompFinite S M P (InputCompositionPolynomialScalar.chart (X := X) plus)
        (InputCompositionPolynomialScalar.laurentReference f g plus)
        (InputCompositionPolynomialScalar.anchor f g plus)
        (InputCompositionPolynomialScalar.anchor f g plus) (y, Convexity.StdSimplex.single 4)) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (InputCompositionProjectionAttachment.plusConeMap f g y) :=
  terminal_carrier f g c plus plus_projection _ (ProjectionReference.FourEdge.beta_plus X E)
    _ (ProjectionReference.FourEdge.beta_plus X (A ≫ E)) (terminal_plus_cylinder f g c) y

/-- The distinct negative source-square carrier obeys the same actual full scalar seam. -/
theorem terminal_minus_carrier (y : InputCompositionPolynomialScalar.Carrier f g minus) :
    preMinusConeMap f g c
      (FinitePrecomp.precompFinite S M P (InputCompositionPolynomialScalar.chart (X := X) minus)
        (InputCompositionPolynomialScalar.laurentReference f g minus)
        (InputCompositionPolynomialScalar.anchor f g minus)
        (InputCompositionPolynomialScalar.anchor f g minus) (y, Convexity.StdSimplex.single 4)) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (InputCompositionProjectionAttachment.minusConeMap f g y) :=
  terminal_carrier f g c minus minus_projection _ (ProjectionReference.FourEdge.beta_minus X E)
    _ (ProjectionReference.FourEdge.beta_minus X (A ≫ E)) (terminal_minus_cylinder f g c) y
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
