attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
  (k : laurent ⟶ polynomial)
  (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "G" => InputCompositionPolynomialScalar.laurentReference f g k
local notation "K" => InputCompositionPolynomialScalar.chart (X := X) k
local notation "D" => BassH0Merge.normalizedMap (projection.app W)
local notation "M" => BassH0Merge.normalizedMap (polynomial.map c)
local notation "ε" => InputCompositionPolynomialScalar.anchor f g k
local notation "εM" => CutOpenPrecomp.anchor M P K G ε

/-- Every genuine polynomial-source node attaches through the actual outer projection. -/
theorem initialProjection_attachment_node (N : Node (M ≫ P)) :
    reanchor (initialLaurentProjectionAnchor f g c k hk)
      (precompNode D (attachPostcompNode (M ≫ P) K (M ≫ G) εM N)) =
      ProjectionReference.chartNode X k hk (A ≫ E)
        (reanchor (initialPolynomialProjectionAnchor f g c) (precompNode D N)) := by
  rw [CutOpenPrecomp.attachment_node D (M ≫ P) K (M ≫ G) εM N]
  apply Sigma.ext
  · rfl
  apply heq_of_eq
  change initialLaurentProjectionAnchor f g c k hk ≪≫
      (CutOpenPrecomp.anchor D (M ≫ P) K (M ≫ G) εM ≪≫
        isoWhiskerRight (precompNode D N).2 (K).functor) =
    ProjectionReference.chartAnchor X k hk (A ≫ E) ≪≫
      isoWhiskerRight (initialPolynomialProjectionAnchor f g c ≪≫ (precompNode D N).2) (K).functor
  rw [isoWhiskerRight_trans, ← Iso.trans_assoc, initialProjection_attachment, Iso.trans_assoc]
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
