open CommonReferenceNerve
attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

open BassReferenceMerge

/-- The terminal source row uses the actual projection naturality equality. -/
theorem middleSourceTotal :
    laurent.map c ≫ laurentProjection.app W = (plus.app Z ≫ projection.app Z) ≫ c := by
  rw [plus_projection Z, laurentProjection.naturality c]
  rfl

/-- The complete terminal row anchor in normalized two-factor form. -/
theorem middleSourceAnchor : ProjectionReference.Target.terminalBaseAnchor c =
    smallIso (eqToIso (congrArg boundedFlatNormalizedPullback (middleSourceTotal c)) ≪≫
      boundedFlatNormalizedPullbackCompIso (plus.app Z ≫ projection.app Z) c) := by
  rw [smallIso_trans, smallIso_eqToIso]
  rfl

/-- The two full normalized three-factor routes agree before cancelling the common prefix. -/
theorem middleThreeFactor :
    BinaryNormalizationSimplex.total (laurent.map (f ≫ g) ≫ laurentProjection.app Z) c ≪≫
      isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
        (InputCompositionCarrierBottom.sourceIso f g) =
    smallIso (eqToIso (congrArg boundedFlatNormalizedPullback
      (lowerLaurent_total f g c).symm)) ≪≫
      BinaryNormalizationSimplex.total (laurent.map f ≫ laurent.map g)
        (laurent.map c ≫ laurentProjection.app W) ≪≫
      isoWhiskerRight (ProjectionReference.Target.terminalBaseAnchor c)
        (lowerLaurentBinary f g).functor := by
  rw [middleSourceAnchor]
  change smallIso (boundedFlatNormalizedPullbackCompIso
      (laurent.map (f ≫ g) ≫ laurentProjection.app Z) c) ≪≫
    isoWhiskerLeft (AsSmall.down ⋙ boundedFlatNormalizedPullback c ⋙ AsSmall.up)
      (smallIso (eqToIso (congrArg boundedFlatNormalizedPullback
        (InputCompositionCarrierBottom.total f g).symm) ≪≫
        boundedFlatNormalizedPullbackCompIso (laurent.map f ≫ laurent.map g)
          (plus.app Z ≫ projection.app Z))) =
    smallIso (eqToIso (congrArg boundedFlatNormalizedPullback
      (lowerLaurent_total f g c).symm)) ≪≫
      smallIso (boundedFlatNormalizedPullbackCompIso (laurent.map f ≫ laurent.map g)
        (laurent.map c ≫ laurentProjection.app W)) ≪≫
      isoWhiskerRight (smallIso (eqToIso (congrArg boundedFlatNormalizedPullback
        (middleSourceTotal c)) ≪≫
        boundedFlatNormalizedPullbackCompIso (plus.app Z ≫ projection.app Z) c))
        (AsSmall.down ⋙ boundedFlatNormalizedPullback
          (laurent.map f ≫ laurent.map g) ⋙ AsSmall.up)
  rw [smallIso_precomp, smallIso_postcomp]
  simp only [← smallIso_trans]
  apply congrArg smallIso
  simpa only [eqToIso_refl, Iso.refl_trans, Iso.trans_assoc] using
    (anchored_factorizations_strict (laurent.map f ≫ laurent.map g)
      (plus.app Z ≫ projection.app Z) c
      (laurent.map (f ≫ g) ≫ laurentProjection.app Z)
      (laurent.map c ≫ laurentProjection.app W)
      ((laurent.map (f ≫ g) ≫ laurentProjection.app Z) ≫ c)
      (InputCompositionCarrierBottom.total f g).symm (middleSourceTotal c) rfl
      (lowerLaurent_total f g c).symm)

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
