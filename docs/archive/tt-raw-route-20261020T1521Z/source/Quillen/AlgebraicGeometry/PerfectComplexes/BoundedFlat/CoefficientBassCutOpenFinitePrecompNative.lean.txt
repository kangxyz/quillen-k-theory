@[expose] public section

open CategoryTheory Convexity

universe u
namespace Quillen.Waldhausen.IteratedS.CommonReferenceNerve.FinitePrecomp
open CutOpenCarrier
variable {U V W T : Model.{u}} (A : U ⟶ V)
variable {q : ℕ} (M : Fin (q + 1) → Node A)
variable (F : V ⟶ W) (J : W ⟶ T) (G : V ⟶ T)
variable (ε₀ ε₁ : G.functor ≅ (F ≫ J).functor)

/-- Every native input follows the finite source family before the original carrier evaluator. -/
theorem precompFinite_native (n : ℕ) (y : space F J G ε₀ ε₁)
    (w : StdSimplex ℝ (Fin (q + 1))) (x : (prespectrum U).space n) :
    nativeEval (A ≫ F) J (A ≫ G)
      (CutOpenPrecomp.anchor A F J G ε₀) (CutOpenPrecomp.anchor A F J G ε₁) n
      (precompFinite A M F J G ε₀ ε₁ (y, w)) x =
    nativeEval F J G ε₀ ε₁ n y (prespectrumFamily A M n (w, x)) := by
  rcases exists_native_cell F J G ε₀ ε₁ y with
    ⟨z, rfl⟩ | ⟨z, t, rfl⟩ | ⟨z, t, rfl⟩ | rfl | rfl
  · rw [precompFinite_laurent, nativeEval_laurentInclusion, nativeEval_laurentInclusion,
      parameter_native]
  · rw [precompFinite_cylinderMinus, nativeEval_cylinderMinus, nativeEval_cylinderMinus,
      parameter_native]
  · rw [precompFinite_cylinderPlus, nativeEval_cylinderPlus, nativeEval_cylinderPlus,
      parameter_native]
  · rw [precompFinite_tipMinus, nativeEval_tipMinus, nativeEval_tipMinus]
  · rw [precompFinite_tipPlus, nativeEval_tipPlus, nativeEval_tipPlus]

end Quillen.Waldhausen.IteratedS.CommonReferenceNerve.FinitePrecomp
