/-- Polynomial Square left for the exact original reference and parameter data. -/
theorem polynomialSquare_left (r : unitInterval) :
    InputCompositionSpanScalar.polynomialSquare f g (r, 0) =
      mapParameter (polynomialNode f g)
        (InputCompositionPolynomialScalar.polynomialPath f g r) := by
  let h : ⦋2⦌ ⟶ ⦋17⦌ := InputCompositionTopChartSeams.movingEdge
  change parameter T _ (DoubleSquareScalarPaths.coefficientMovingSquareParameter (r, 0)) = _
  rw [CoefficientMovingSquareRows.moving_left_row]
  change parameter T _ (StdSimplex.map h.toOrderHom (binaryPath (k := 2) 0 1 2 r)) =
    mapParameter (polynomialNode f g)
      (parameter P (InputCompositionPolynomialScalar.polynomialNodes f g) (binaryPath 0 1 2 r))
  rw [parameter_face]
  exact (congrArg (fun N ↦ parameter T N (binaryPath (k := 2) 0 1 2 r))
    (funext (polynomial_target_nodes f g))).trans
      (mapParameter_parameter (polynomialNode f g)
        (InputCompositionPolynomialScalar.polynomialNodes f g) (binaryPath 0 1 2 r)).symm
