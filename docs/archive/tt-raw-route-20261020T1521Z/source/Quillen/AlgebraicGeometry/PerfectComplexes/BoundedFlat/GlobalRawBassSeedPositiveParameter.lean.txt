/-!
# The original positive middle seam in the full Laurent nerve

The postcomposed K path and precomposed D path have the same three complete reference
nodes. Their realized parameters therefore agree with the original binary clocks,
before native evaluation or passage to the cut-open carrier.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Convexity
open Quillen.Waldhausen.IteratedS Quillen.SimplexParameters
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open scoped Simplicial Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.SeedPositiveSeam
open CommonReferenceNerve
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- The original K path selects the last three positive rectangle vertices. -/
private def firstFace : ⦋2⦌ ⟶ ⦋17⦌ :=
  SimplexCategory.Hom.mk ⟨![15, 16, 17], by
    intro i j hij; fin_cases i <;> fin_cases j <;> simp_all⟩

/-- The original D path selects the three source-chart rectangle vertices. -/
private def secondFace : ⦋2⦌ ⟶ ⦋17⦌ :=
  SimplexCategory.Hom.mk ⟨![2, 3, 11], by
    intro i j hij; fin_cases i <;> fin_cases j <;> simp_all⟩

/-- The full postcomposed K path and precomposed D path agree before native evaluation. -/
theorem positive_parameter (s : I) :
    mapParameter (InputCompositionTerminalAttach.postNode f g)
      (parameter (InputDescSimplex.reference g) (InputDescSimplex.nodes g)
        (ProjectionRectangleParameters.K BassH0Diagram.plusFace s)) =
    mapParameter (InputCompositionTerminalAttach.preNode f g)
      (parameter (InputDescSimplex.reference f) (InputDescSimplex.nodes f)
        (ProjectionRectangleParameters.D BassH0Diagram.plusFace s)) := by
  rw [mapParameter_parameter, mapParameter_parameter]
  change parameter _ _
      (binaryPath ((firstFace ≫ BassH0Diagram.plusFace).toOrderHom 0)
        ((firstFace ≫ BassH0Diagram.plusFace).toOrderHom 1)
        ((firstFace ≫ BassH0Diagram.plusFace).toOrderHom 2) s) =
    parameter _ _
      (binaryPath ((secondFace ≫ BassH0Diagram.plusFace).toOrderHom 0)
        ((secondFace ≫ BassH0Diagram.plusFace).toOrderHom 1)
        ((secondFace ≫ BassH0Diagram.plusFace).toOrderHom 2) s)
  rw [← map_binaryPath (firstFace ≫ BassH0Diagram.plusFace) 0 1 2 s,
    ← map_binaryPath (secondFace ≫ BassH0Diagram.plusFace) 0 1 2 s]
  change parameter _ _ (StdSimplex.map (firstFace ≫ BassH0Diagram.plusFace).toOrderHom _) =
    parameter _ _ (StdSimplex.map (secondFace ≫ BassH0Diagram.plusFace).toOrderHom _)
  rw [parameter_face, parameter_face]
  apply congrArg (fun N ↦ parameter (InputIdentitySeedCarrier.laurentRef (f ≫ g)) N
    (binaryPath (k := 2) 0 1 2 s))
  funext i
  fin_cases i
  · exact InputCompositionTerminalMiddle.plus_nodes f g 0
  · exact InputCompositionTerminalMiddle.plus_nodes f g 1
  · exact InputCompositionTerminalMiddle.plus_nodes f g 2

end Quillen.PerfectComplexes.GlobalRawBass.SeedPositiveSeam
