/-- Every genuine polynomial lower-half input follows the original chart square. -/
theorem polynomialLower_native (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj W))).space n) :
    nativeEval (MC ≫ P) K (MC ≫ G) ε' ε' n (polynomialLower f g c k (r, s)) x =
      (Quillen.LoopPrespectrum.mappingConeInclusion
        (InputCompositionPolynomialScalar.chartP k X)).map n
        ((InputCompositionCharts.laurentComposition f g).level n
          (r, (boundedFlatPerfectPrespectrumSquareHomotopy (laurent.map c) (k.app W)
            (k.app Z) (polynomial.map c) (k.naturality c)).level n (s, x))) := by
  change nativeEval (MC ≫ P) K (MC ≫ G) ε' ε' n
    (laurentInclusion (MC ≫ P) K (MC ≫ G) ε' ε' (anchoredLowerChart f g c k (r, s))) x = _
  rw [nativeEval_laurentInclusion, anchoredLowerChart_native]
  rfl

/-- The entire genuine polynomial upper half retains the selected original corrected square. -/
theorem polynomialUpper_native (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj W))).space n) :
    nativeEval (MC ≫ P) K (MC ≫ G) ε' ε' n (polynomialUpper f g c k (r, s)) x =
      (InputCompositionPolynomialScalar.nullSquare f g k).level n
        (r, (s, (boundedFlatPerfectPrespectrumMap (polynomial.map c)).map n x)) := by
  exact (CutOpenPrecomp.map_native MC P K G ε ε n
    ((InputCompositionPolynomialScalar.correctedSquare f g k).parameter (r, s)) x).trans
      (InputCompositionPolynomialScalar.correctedSquare_native f g k n r s _)
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
