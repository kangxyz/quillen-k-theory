variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
  (k : laurent ⟶ polynomial)
  (hk : ∀ W, k.app W ≫ projection.app W = laurentProjection.app W)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "J" => ProjectionReference.projectionMap X
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "G" => InputCompositionPolynomialScalar.laurentReference f g k

local notation "L" => ProjectionReference.reference X E

local notation "β" => ProjectionReference.FourEdge.beta X E

    (InputCompositionTopPieces.baseSource f g ≫ ProjectionReference.projectionMap X) ⟶
      FourEdgeTree.simplicial (InputCompositionTopPieces.baseSource f g)
        (ProjectionReference.projectionMap X))
  (hv : v ≫ ProjectionReference.FourEdge.beta X (InputCompositionTopPieces.baseSource f g) =
    ProjectionReference.FourEdge.chartMap X (InputCompositionTopPieces.baseSource f g) k hk)
include hv in
/-- Cylinder Map zero for the exact original reference and parameter data. -/
theorem cylinderMap_zero (z : |parameterSSet P|) :
    cylinderMap f g v (z, 0) =
      laurentMap f g k hk (mapParameter (attachPostcompNode P K G ε) z) := by
  change FourEdgeTree.cylinder E J L β (_, 0) = _
  rw [FourEdgeTree.cone_cylinder_zero, ← FourEdgeTree.realized_comp, hv]
  apply congrArg (FourEdgeTree.laurentInclusion E J L β)
  obtain ⟨n, N, w, rfl⟩ := exists_parameter P z
  change mapParameter (ProjectionReference.chartNode X k hk E)
      (mapParameter (polynomialNode f g) (parameter P N w)) =
    mapParameter (laurentNode f g k hk)
      (mapParameter (attachPostcompNode P K G ε) (parameter P N w))
  rw [mapParameter_parameter, mapParameter_parameter,
    mapParameter_parameter, mapParameter_parameter]
  exact congrArg (fun ns ↦ parameter L ns w)
    (funext fun i ↦ (chart_attachment_node f g k hk (N i)).symm)

/-- Cylinder Map one for the exact original reference and parameter data. -/
theorem cylinderMap_one (z : |parameterSSet P|) :
    cylinderMap f g v (z, 1) = FourEdgeTree.tip E J L β :=
  FourEdgeTree.cone_cylinder_one E J L β _

/-- Cone Map for the exact original reference and parameter data. -/
def coneMap : C(InputCompositionPolynomialScalar.Carrier f g k,
    ProjectionReference.FourEdge.scalarCone X E) :=
  CutOpenFold.map P K G ε (ProjectionReference.FourEdge.scalarCone X E)
    (laurentMap f g k hk) (cylinderMap f g v) (FourEdgeTree.tip E J L β)
    (cylinderMap_zero f g k hk v hv) (cylinderMap_one f g v)

/-- Cone Map laurent for the exact original reference and parameter data. -/
theorem coneMap_laurent (z : |parameterSSet G|) :
    coneMap f g k hk v hv (CutOpenCarrier.laurentInclusion P K G ε ε z) =
      laurentMap f g k hk z :=
  CutOpenFold.map_laurent _ _ _ _ _ _ _ _ _ _ z

/-- Cone Map cylinder for the exact original reference and parameter data. -/
theorem coneMap_cylinder (z : |parameterSSet P|) (s : unitInterval) :
    coneMap f g k hk v hv (CutOpenCarrier.cylinderMinus P K G ε ε (z, s)) =
      cylinderMap f g v (z, s) :=
  CutOpenFold.map_cylinder _ _ _ _ _ _ _ _ _ _ z s
