/-! The chart-relative whole map recovers the original raw cone map on each genuine sector. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry

universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.ChartRelativeTransport
open CommonReferenceNerve
variable {X Y : Scheme.{u}} (a : X ⟶ Y)
  {V : Model.{u + 1}} (P : V ⟶ modelOf (boundedFlatPerfectWaldhausen (polynomial.obj Y)))
  (G : V ⟶ modelOf (boundedFlatPerfectWaldhausen (laurent.obj Y)))

/-- Exact original raw postcomposition on every plus chart-relative scalar point. -/
theorem plus_raw_native
    (eps : G.functor ≅ (P ≫ InputCompositionTerminalAttach.arrow (plus.app Y)).functor)
    (n : ℕ) (z : sourceCarrier plus P G eps) (x : (prespectrum V).space n) :
    (InputIdentityPositive.inclusion X).map n
      (CutOpenCarrier.nativeEval (targetPolynomial a P)
        (InputCompositionTerminalAttach.arrow (plus.app X)) (targetLaurent a plus P)
        (targetAnchor a plus P) (targetAnchor a plus P) n (transport a plus P G eps z) x) =
      (GlobalRawBass.coneMap a).map n ((InputIdentityPositive.inclusion Y).map n
        (CutOpenCarrier.nativeEval P (InputCompositionTerminalAttach.arrow (plus.app Y))
          G eps eps n z x)) := by
  rw [transport_native]
  exact (congrArg (fun H ↦ H.map n
    (CutOpenCarrier.nativeEval P (InputCompositionTerminalAttach.arrow (plus.app Y))
      G eps eps n z x)) (InputIdentityPositive.coneMap_factor a)).symm

/-- Exact original raw postcomposition on every minus chart-relative scalar point. -/
theorem minus_raw_native
    (eps : G.functor ≅ (P ≫ InputCompositionTerminalAttach.arrow (minus.app Y)).functor)
    (n : ℕ) (z : sourceCarrier minus P G eps) (x : (prespectrum V).space n) :
    (InputIdentityNegative.inclusion X).map n
      (CutOpenCarrier.nativeEval (targetPolynomial a P)
        (InputCompositionTerminalAttach.arrow (minus.app X)) (targetLaurent a minus P)
        (targetAnchor a minus P) (targetAnchor a minus P) n (transport a minus P G eps z) x) =
      (GlobalRawBass.coneMap a).map n ((InputIdentityNegative.inclusion Y).map n
        (CutOpenCarrier.nativeEval P (InputCompositionTerminalAttach.arrow (minus.app Y))
          G eps eps n z x)) := by
  rw [transport_native]
  exact (congrArg (fun H ↦ H.map n
    (CutOpenCarrier.nativeEval P (InputCompositionTerminalAttach.arrow (minus.app Y))
      G eps eps n z x)) (InputIdentityNegative.coneMap_factor a)).symm
end Quillen.PerfectComplexes.GlobalRawBass.ChartRelativeTransport
