module
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassPrewhiskerTerminalCarrier
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassInputCompositionProjectionConeCorrected
/-!
The original overlap interior is precomposed by the actual fixed source map.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Quillen.Waldhausen.IteratedS
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open scoped Simplicial Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c

/-- The original overlap interior is precomposed by the actual fixed source map. -/
def selectedOverlap := (ProjectionReference.FourEdge.Precomp.cone X A E).comp
  (InputCompositionOverlapScalar.correctedFamily f g)

/-- The complete chosen positive square joins the original overlap at its unchanged endpoint. -/
theorem selectedPlus_one (r s : I) :
    selectedPlus f g c ((r, s), 1) = selectedOverlap f g c ((r, s), 0) := by
  have h := terminal_plus_carrier f g c
    ((InputCompositionPolynomialScalar.correctedSquare f g plus).parameter (r, s))
  have he : selectedPlus f g c ((r, s), 1) =
      ProjectionReference.FourEdge.Precomp.cone X A E
        (InputCompositionProjectionAttachment.plusConeMap f g
          ((InputCompositionPolynomialScalar.correctedSquare f g plus).parameter (r, s))) := by
    simpa only [selectedPlus, selectedPolynomial, ContinuousMap.comp_apply, ContinuousMap.coe_mk,
      Path.target] using h
  exact he.trans (congrArg (ProjectionReference.FourEdge.Precomp.cone X A E)
    (InputCompositionProjectionAttachment.plus_corrected f g r s))

/-- The distinct negative endpoint joins the other end of the same original overlap interior. -/
theorem selectedMinus_one (r s : I) :
    selectedMinus f g c ((r, s), 1) = selectedOverlap f g c ((r, s), 1) := by
  have h := terminal_minus_carrier f g c
    ((InputCompositionPolynomialScalar.correctedSquare f g minus).parameter (r, s))
  have he : selectedMinus f g c ((r, s), 1) =
      ProjectionReference.FourEdge.Precomp.cone X A E
        (InputCompositionProjectionAttachment.minusConeMap f g
          ((InputCompositionPolynomialScalar.correctedSquare f g minus).parameter (r, s))) := by
    simpa only [selectedMinus, selectedPolynomial, ContinuousMap.comp_apply, ContinuousMap.coe_mk,
      Path.target] using h
  exact he.trans (congrArg (ProjectionReference.FourEdge.Precomp.cone X A E)
    (InputCompositionProjectionAttachment.minus_corrected f g r s))
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
