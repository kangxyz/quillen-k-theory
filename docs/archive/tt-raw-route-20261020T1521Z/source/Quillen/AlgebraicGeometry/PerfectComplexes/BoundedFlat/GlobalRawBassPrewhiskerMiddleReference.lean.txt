attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

private theorem cancelSix {C : Type*} [Category C] {a b c d e k : C}
    (α : a ≅ b) (β : c ≅ b) (γ : c ≅ d) (δ : d ≅ e) (ε : e ≅ k) (tau : b ≅ k)
    (h : β ≪≫ tau = γ ≪≫ δ ≪≫ ε) :
    (α ≪≫ β.symm ≪≫ γ ≪≫ δ) ≪≫ ε = α ≪≫ tau := by
  simp only [Iso.trans_assoc]
  rw [← h, Iso.symm_self_id_assoc]

/-- The original lower reference and the fixed-source bottom grid have the same full anchor. -/
theorem middleReference :
    lowerLaurentAnchor f g c ≪≫
      isoWhiskerRight (ProjectionReference.Target.terminalBaseAnchor c)
        (lowerLaurentBinary f g).functor ≪≫
      Functor.associator (BassH0Merge.normalizedMap c).functor
        (InputIdentityProjection.source Z).functor (lowerLaurentBinary f g).functor =
    Functor.associator (BassH0Merge.normalizedMap c).functor
        (InputCompositionTopPieces.baseSource f g).functor
        (InputIdentityProjection.source X).functor ≪≫
      isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
        ((InputCompositionTopPieces.middleAnchor f g).symm ≪≫
          InputCompositionCarrierBottom.sourceIso f g) := by
  have h := cancelSix
    (isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
      (InputCompositionTopPieces.middleAnchor f g).symm)
    (BinaryNormalizationSimplex.total
      (laurent.map (f ≫ g) ≫ laurentProjection.app Z) c)
    (BassReferenceMerge.smallIso (eqToIso (congrArg boundedFlatNormalizedPullback
      (lowerLaurent_total f g c).symm)))
    (BinaryNormalizationSimplex.total (laurent.map f ≫ laurent.map g)
      (laurent.map c ≫ laurentProjection.app W))
    (isoWhiskerRight (ProjectionReference.Target.terminalBaseAnchor c)
      (lowerLaurentBinary f g).functor)
    (isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
      (InputCompositionCarrierBottom.sourceIso f g))
    (middleThreeFactor f g c)
  exact (congrArg (fun e ↦ lowerLaurentAnchor f g c ≪≫ e)
    (middleTerminalAssociator f g c)).trans
    ((congrArg (fun e ↦ e ≪≫ isoWhiskerRight
      (ProjectionReference.Target.terminalBaseAnchor c) (lowerLaurentBinary f g).functor)
      (middleLeftIdentity f g c)).trans (h.trans (middleRightIdentity f g c).symm))
