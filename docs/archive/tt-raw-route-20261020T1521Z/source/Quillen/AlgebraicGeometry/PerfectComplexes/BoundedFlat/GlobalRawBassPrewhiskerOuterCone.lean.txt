variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
  (k : laurent ⟶ polynomial)
  (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "S" => BassH0Merge.normalizedMap (polynomial.map c)
local notation "D" => BassH0Merge.normalizedMap (projection.app W)
local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "G" => InputCompositionPolynomialScalar.laurentReference f g k
local notation "K" => InputCompositionPolynomialScalar.chart (X := X) k
local notation "ε" => InputCompositionPolynomialScalar.anchor f g k
local notation "ε'" => CutOpenPrecomp.anchor S P K G ε
local notation "J" => ProjectionReference.projectionMap X
local notation "L" => ProjectionReference.reference X (A ≫ E)
local notation "β" => ProjectionReference.FourEdge.beta X (A ≫ E)
local notation "Q" => ProjectionReference.FourEdge.scalarCone X (A ≫ E)

/-- The genuine polynomial-source precomposition carrier, with both original attachments. -/
abbrev outerPolynomialCarrier := CutOpenCarrier.space (S ≫ P) K (S ≫ G) ε' ε'

/-- The full Laurent nerve uses the actual initial source-projection anchor. -/
def outerLaurentMap : C(|parameterSSet (S ≫ G)|, Q) :=
  (FourEdgeTree.laurentInclusion (A ≫ E) J L β).comp
    (mapParameter (fun N ↦ reanchor (initialLaurentProjectionAnchor f g c k hk) (precompNode D N)))

variable (v : parameterSSet ((BassH0Merge.normalizedMap c ≫
    InputCompositionTopPieces.baseSource f g) ≫ ProjectionReference.projectionMap X) ⟶
    FourEdgeTree.simplicial (BassH0Merge.normalizedMap c ≫
      InputCompositionTopPieces.baseSource f g) (ProjectionReference.projectionMap X))
  (hv : v ≫ ProjectionReference.FourEdge.beta X
      (BassH0Merge.normalizedMap c ≫ InputCompositionTopPieces.baseSource f g) =
    ProjectionReference.FourEdge.chartMap X
      (BassH0Merge.normalizedMap c ≫ InputCompositionTopPieces.baseSource f g) k hk)

/-- Reanchor each entire polynomial node and retain its independent original cone height. -/
def outerCylinderMap : C(|parameterSSet (S ≫ P)| × I, Q) :=
  ⟨fun z ↦ FourEdgeTree.cylinder (A ≫ E) J L β
    (SSet.toTop.map v (mapParameter (fun N ↦ reanchor (initialPolynomialProjectionAnchor f g c)
      (precompNode D N)) z.1), z.2),
    by fun_prop⟩

include hv in
/-- The zero boundary is the full scalar attachment, proved before evaluation. -/
theorem outerCylinderMap_zero (z : |parameterSSet (S ≫ P)|) :
    outerCylinderMap f g c v (z, 0) =
      outerLaurentMap f g c k hk (mapParameter (attachPostcompNode (S ≫ P) K (S ≫ G) ε') z) := by
  change FourEdgeTree.cylinder (A ≫ E) J L β (_, 0) = _
  rw [FourEdgeTree.cone_cylinder_zero, ← FourEdgeTree.realized_comp, hv]
  apply congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L β)
  obtain ⟨n, N, w, rfl⟩ := exists_parameter (S ≫ P) z
  change mapParameter (ProjectionReference.chartNode X k hk (A ≫ E))
      (mapParameter (fun N ↦ reanchor (initialPolynomialProjectionAnchor f g c) (precompNode D N))
        (parameter (S ≫ P) N w)) =
    mapParameter (fun N ↦ reanchor (initialLaurentProjectionAnchor f g c k hk) (precompNode D N))
      (mapParameter (attachPostcompNode (S ≫ P) K (S ≫ G) ε') (parameter (S ≫ P) N w))
  have h1 := mapParameter_parameter (F := S ≫ P) («G» := (A ≫ E) ≫ J)
    (fun N ↦ reanchor (initialPolynomialProjectionAnchor f g c) (precompNode D N)) N w
  have h2 := mapParameter_parameter (F := (A ≫ E) ≫ J) («G» := L)
    (ProjectionReference.chartNode X k hk (A ≫ E))
    (fun i ↦ reanchor (initialPolynomialProjectionAnchor f g c) (precompNode D (N i))) w
  have h3 := mapParameter_parameter (F := S ≫ P) («G» := S ≫ G)
    (attachPostcompNode (S ≫ P) K (S ≫ G) ε') N w
  have h4 := mapParameter_parameter (F := S ≫ G) («G» := L)
    (fun N ↦ reanchor (initialLaurentProjectionAnchor f g c k hk) (precompNode D N))
    (fun i ↦ attachPostcompNode (S ≫ P) K (S ≫ G) ε' (N i)) w
  exact ((congrArg (mapParameter (ProjectionReference.chartNode X k hk (A ≫ E))) h1).trans h2).trans
    ((congrArg (fun ns ↦ parameter L ns w)
      (funext fun i ↦ (initialProjection_attachment_node f g c k hk (N i)).symm)).trans
      ((congrArg (mapParameter (fun N ↦ reanchor (initialLaurentProjectionAnchor f g c k hk)
        (precompNode D N))) h3).trans h4).symm)

/-- The actual terminal cone tip is unchanged. -/
theorem outerCylinderMap_one (z : |parameterSSet (S ≫ P)|) :
    outerCylinderMap f g c v (z, 1) = FourEdgeTree.tip (A ≫ E) J L β :=
  FourEdgeTree.cone_cylinder_one (A ≫ E) J L β _

/-- The entire genuine polynomial-source chart carrier descends to the common overlap scalar cone.
-/
def outerConeMap : C(outerPolynomialCarrier f g c k, Q) :=
  CutOpenFold.map (S ≫ P) K (S ≫ G) ε' Q (outerLaurentMap f g c k hk)
    (outerCylinderMap f g c v) (FourEdgeTree.tip (A ≫ E) J L β)
    (outerCylinderMap_zero f g c k hk v hv) (outerCylinderMap_one f g c v)

/-- The positive polynomial-source carrier lands in the original positive tree vertex. -/
def outerPlusConeMap := outerConeMap f g c plus plus_projection
  (FourEdgeTree.plusVertex (A ≫ E) J) (ProjectionReference.FourEdge.beta_plus X (A ≫ E))

/-- The negative polynomial-source carrier keeps its distinct original tree vertex. -/
def outerMinusConeMap := outerConeMap f g c minus minus_projection
  (FourEdgeTree.minusVertex (A ≫ E) J) (ProjectionReference.FourEdge.beta_minus X (A ≫ E))
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
