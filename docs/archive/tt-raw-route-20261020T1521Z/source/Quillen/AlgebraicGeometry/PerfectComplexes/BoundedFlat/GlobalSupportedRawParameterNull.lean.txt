/-!
# The original geometric localization zero-null on literal parameter rows

The actual right-factor product rows retain their independently normalized geometric pullbacks.
Their chosen-zero transformation is defined without conjugating through an equivalence. The
whole section factor-swap recovery proves its weakness, and the actual evaluation triangle
compares both complete nulls. This supplies polynomial and Laurent parameter instances without
identifying their products with parameter-first pairs by definitional equality.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory CategoryTheory.Functor AlgebraicGeometry Opposite
open Quillen.Waldhausen.IteratedS
open scoped Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalSupportedRawBass.Localization.Parameter
open GlobalSectionChangeIndex GlobalSectionChangeIndex.SupportedParameterSwap
open SupportedTensorEvaluation
open AlgebraicGeometry.Scheme.Products AlgebraicGeometry.Scheme.Bass
variable {U X S : Scheme.{u}} (j : U ⟶ X) [IsOpenImmersion j]
  [CompactSpace U] [CompactSpace X] (T : X.Opens)
  (p : S ⟶ integerBase) [QuasiCompact p]

/-- The complete literal right-factor source pair. -/
abbrev sourcePair : Scheme.OpenPairs.{u} :=
  ((pairBaseProductFunctor p).obj (Rows.source j T)).obj

/-- The complete literal right-factor middle pair. -/
abbrev middlePair : Scheme.OpenPairs.{u} :=
  ((pairBaseProductFunctor p).obj (Rows.middle T)).obj

/-- The complete literal right-factor target pair. -/
abbrev targetPair : Scheme.OpenPairs.{u} :=
  ((pairBaseProductFunctor p).obj (Rows.target j T)).obj

/-- The unchanged right-factor enlargement pair arrow. -/
abbrev enlargementPair : middlePair T p ⟶ sourcePair j T p :=
  ((pairBaseProductFunctor p).map (Rows.enlargement j T)).hom

/-- The unchanged right-factor restriction pair arrow. -/
abbrev restrictionPair : targetPair j T p ⟶ middlePair T p :=
  ((pairBaseProductFunctor p).map (Rows.restriction j T)).hom

/-- The actual source geometric supported Model. -/
abbrev sourceModel : Model.{u+1} := rawModel (sourcePair j T p)

/-- The actual intermediate geometric supported Model. -/
abbrev middleModel : Model.{u+1} := rawModel (middlePair T p)

/-- The actual target geometric supported Model. -/
abbrev targetModel : Model.{u+1} := rawModel (targetPair j T p)

/-- The original normalized support-enlargement pullback. -/
def rawEnlargement : sourceModel j T p ⟶ middleModel T p :=
  evaluationRawMap (enlargementPair j T p)

/-- The original normalized supported open restriction. -/
def rawRestriction : middleModel T p ⟶ targetModel j T p :=
  evaluationRawMap (restrictionPair j T p)

/-- The raw composite is defined only from the two original geometric pullbacks. -/
def rawComposite : sourceModel j T p ⟶ targetModel j T p :=
  rawEnlargement j T p ≫ rawRestriction j T p

/-- The unchanged normalized evaluation on the union-supported source. -/
abbrev sourceEvaluation : strictSource j T p ⟶ sourceModel j T p :=
  evaluationModel (X := op (sourcePair j T p).obj.right) (sourcePair j T p).obj.hom

/-- The unchanged normalized evaluation on the inverse-image-supported target. -/
abbrev targetEvaluation : strictTarget j T p ⟶ targetModel j T p :=
  evaluationModel (X := op (targetPair j T p).obj.right) (targetPair j T p).obj.hom

/-- Paste the two actual evaluation comparisons, retaining every original associator. -/
def evaluationIso :
    (strictComposite j T p ≫ targetEvaluation j T p).functor ≅
      (sourceEvaluation j T p ≫ rawComposite j T p).functor :=
  pasteModelSquares (strictEnlargement j T p) (strictRestriction j T p)
    (rawEnlargement j T p) (rawRestriction j T p)
    (sourceEvaluation j T p)
    (evaluationModel (X := op (middlePair T p).obj.right) (middlePair T p).obj.hom)
    (targetEvaluation j T p)
    (evaluationComparisonIso (enlargementPair j T p))
    (evaluationComparisonIso (restrictionPair j T p))

/-- The geometric zero-to-composite map is independent of the evaluation equivalence. -/
def zeroToRawFull :
    (zeroHom (sourceModel j T p) (targetModel j T p)).functor ⟶ (rawComposite j T p).functor :=
  ModelZeroNormalization.zeroToMap (rawComposite j T p)

/-- The old support-acyclicity and the actual evaluation counit prove every raw component weak. -/
theorem zeroToRawFull_weak (Y : sourceModel j T p) :
    (targetModel j T p).waldhausen.weq ((zeroToRawFull j T p).app Y) :=
  ModelZeroNormalization.zeroToMap_weak_of_comparison
    (strictComposite j T p) (rawComposite j T p) (sourceEvaluation j T p) (targetEvaluation j T p)
    (evaluationIso j T p) (zeroToStrictFull j T p) (zeroToStrictFull_weak j T p)
    (evaluationInverseModel (X := op (sourcePair j T p).obj.right)
      (sourcePair j T p).obj.hom).functor
    (evaluationInverseCounit (X := op (sourcePair j T p).obj.right) (sourcePair j T p).obj.hom) Y

/-- The raw null reverses precisely the independent geometric chosen-zero transformation. -/
def rawNull :=
  prespectrumCompositeZeroNullHomotopy (rawEnlargement j T p) (rawRestriction j T p)
    (zeroToRawFull j T p) (zeroToRawFull_weak j T p)

/-- Every raw null time is its unchanged reversed weak-transformation time. -/
theorem rawNull_level (n : ℕ) (s : unitInterval)
    (x : (prespectrum (sourceModel j T p)).space n) :
    (rawNull j T p).level n (s, x) =
      (prespectrumTransformationHomotopy (zeroToRawFull j T p) (zeroToRawFull_weak j T p)).level n
        (unitInterval.symm s, x) := rfl

/-- The original section null after the actual target evaluation. -/
def postSectionNull :=
  ((strictNull j T p).postcomp (prespectrumMap (targetEvaluation j T p))).cast rfl
    (LoopPrespectrum.zeroHom_comp _)

/-- The geometric null after the actual source evaluation. -/
def preRawNull :=
  (LoopPrespectrum.Homotopy.precomp (prespectrumMap (sourceEvaluation j T p)) (rawNull j T p)).cast
    rfl (LoopPrespectrum.comp_zeroHom _)

/-- The single pasted-isomorphism track has the original composite prespectrum endpoints. -/
def evaluationTrack :=
  (ModelZeroNormalization.comparisonTrack (strictComposite j T p) (rawComposite j T p)
    (sourceEvaluation j T p) (targetEvaluation j T p) (evaluationIso j T p)).cast
      (congrArg (fun k ↦ k ≫ prespectrumMap (targetEvaluation j T p))
        (prespectrumFunctor.map_comp (strictEnlargement j T p) (strictRestriction j T p)))
      (congrArg (fun k ↦ prespectrumMap (sourceEvaluation j T p) ≫ k)
        (prespectrumFunctor.map_comp (rawEnlargement j T p) (rawRestriction j T p)))

/-- The entire original nulls and normalized row edge form a genuine moving square. -/
def evaluationSquare :
    (postSectionNull j T p).Square (preRawNull j T p) (evaluationTrack j T p)
      (LoopPrespectrum.Homotopy.refl
        (LoopPrespectrum.zeroHom (prespectrum (strictSource j T p))
          (prespectrum (targetModel j T p)))) := by
  let S := ModelZeroNormalization.normalizationSquare
    (strictComposite j T p) (rawComposite j T p) (sourceEvaluation j T p) (targetEvaluation j T p)
    (evaluationIso j T p) (zeroToStrictFull j T p) (zeroToRawFull j T p)
    (zeroToStrictFull_weak j T p) (zeroToRawFull_weak j T p)
  exact {
    level := S.level
    at_zero := S.at_zero
    at_one := S.at_one
    left := S.left
    right := S.right
    at_point := S.at_point
    comm := S.comm }

/-- The whole realized weak triangle is the selected evaluation-square family. -/
theorem evaluationSquare_apply (n : ℕ) (r s : unitInterval)
    (x : (prespectrum (strictSource j T p)).space n) :
    (evaluationSquare j T p).level n (r, (s, x)) =
      prespectrumSimplexFamily
        (ModelZeroNormalization.normalizationSimplex
          (strictComposite j T p) (rawComposite j T p)
          (sourceEvaluation j T p) (targetEvaluation j T p)
          (evaluationIso j T p) (zeroToStrictFull j T p) (zeroToStrictFull_weak j T p)) n
        (WeakTriangleSquare.parameter (r, s), x) :=
  ModelZeroNormalization.normalizationSquare_apply
    (strictComposite j T p) (rawComposite j T p) (sourceEvaluation j T p) (targetEvaluation j T p)
    (evaluationIso j T p) (zeroToStrictFull j T p) (zeroToRawFull j T p)
    (zeroToStrictFull_weak j T p) (zeroToRawFull_weak j T p) n r s x

end Quillen.PerfectComplexes.GlobalSupportedRawBass.Localization.Parameter
