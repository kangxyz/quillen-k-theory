public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassInputCompositionProjectionPolynomialEndpoints
/-! Whole first and second homotopy surfaces on both overlap endpoints. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Quillen.Waldhausen.IteratedS
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open scoped Simplicial Topology unitInterval
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.InputCompositionProjectionAttachment
open CommonReferenceNerve CommonReferenceProduct
attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "J" => ProjectionReference.projectionMap X
local notation "L" => ProjectionReference.reference X E
local notation "β" => ProjectionReference.FourEdge.beta X E

/-- Plus Cone Map for the exact original reference and parameter data. -/
def plusConeMap := coneMap f g plus plus_projection (FourEdgeTree.plusVertex E J)
  (ProjectionReference.FourEdge.beta_plus X E)
/-- Minus Cone Map for the exact original reference and parameter data. -/
def minusConeMap := coneMap f g minus minus_projection (FourEdgeTree.minusVertex E J)
  (ProjectionReference.FourEdge.beta_minus X E)

/-- Plus first for the exact original reference and parameter data. -/
theorem plus_first (r s : unitInterval) :
    plusConeMap f g (InputCompositionPolynomialScalar.nullMovingFirst f g plus (r, s)) =
      InputCompositionOverlapScalar.nullMovingFirst f g (r, s) 0 := by
  change coneMap f g plus plus_projection _ _ (CutOpenCarrier.laurentInclusion _ _ _ _ _ _) = _
  rw [coneMap_laurent]
  change FourEdgeTree.laurentInclusion E J L β
    (mapParameter (laurentNode f g plus plus_projection) _) =
      FourEdgeTree.laurentInclusion E J L β
        (InputCompositionSpanScalar.originalOverlapCube f g ![r, s, 0])
  rw [laurent_surface]
  apply congrArg (FourEdgeTree.laurentInclusion E J L β)
  change _ = mapParameter _ (InputCompositionCarrierCube.carrierCube f g ![r, s, 0])
  rw [InputCompositionCarrierCube.cube_left]
  rfl

/-- Minus first for the exact original reference and parameter data. -/
theorem minus_first (r s : unitInterval) :
    minusConeMap f g (InputCompositionPolynomialScalar.nullMovingFirst f g minus (r, s)) =
      InputCompositionOverlapScalar.nullMovingFirst f g (r, s) 1 := by
  change coneMap f g minus minus_projection _ _ (CutOpenCarrier.laurentInclusion _ _ _ _ _ _) = _
  rw [coneMap_laurent]
  change FourEdgeTree.laurentInclusion E J L β
    (mapParameter (laurentNode f g minus minus_projection) _) =
      FourEdgeTree.laurentInclusion E J L β
        (InputCompositionSpanScalar.originalOverlapCube f g ![r, s, 1])
  rw [laurent_surface]
  apply congrArg (FourEdgeTree.laurentInclusion E J L β)
  change _ = mapParameter _ (InputCompositionCarrierCube.carrierCube f g ![r, s, 1])
  rw [InputCompositionCarrierCube.cube_right]
  rfl

/-- Plus second for the exact original reference and parameter data. -/
theorem plus_second (r s : unitInterval) :
    plusConeMap f g (InputCompositionPolynomialScalar.nullMovingSecond f g plus (r, s)) =
      InputCompositionOverlapScalar.nullMovingSecond f g (r, s) 0 := by
  change coneMap f g plus plus_projection _ _ (CutOpenCarrier.cylinderMinus _ _ _ _ _ _) = _
  rw [coneMap_cylinder]
  change FourEdgeTree.cylinder E J L β (_, s) =
    FourEdgeTree.cylinder E J L β (InputCompositionSpanScalar.treeComposition f g (r, 0), s)
  rw [treeComposition_left]

/-- Minus second for the exact original reference and parameter data. -/
theorem minus_second (r s : unitInterval) :
    minusConeMap f g (InputCompositionPolynomialScalar.nullMovingSecond f g minus (r, s)) =
      InputCompositionOverlapScalar.nullMovingSecond f g (r, s) 1 := by
  change coneMap f g minus minus_projection _ _ (CutOpenCarrier.cylinderMinus _ _ _ _ _ _) = _
  rw [coneMap_cylinder]
  change FourEdgeTree.cylinder E J L β (_, s) =
    FourEdgeTree.cylinder E J L β (InputCompositionSpanScalar.treeComposition f g (r, 1), s)
  rw [treeComposition_right]
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionProjectionAttachment
