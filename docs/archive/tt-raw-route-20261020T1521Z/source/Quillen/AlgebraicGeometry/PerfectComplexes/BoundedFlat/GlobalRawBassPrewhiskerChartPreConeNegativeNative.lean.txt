local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c

/-- Every point of the actual negative source-square carrier evaluates by the strict chart
inclusion. -/
theorem preMinusConeMap_native (n : ℕ) (z : sourceSquareCarrier f g c minus)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen W)).space n) :
    ProjectionReference.FourEdge.coneEval X (A ≫ E) n (preMinusConeMap f g c z) x =
      (InputIdentityNegative.inclusion X).map n
        (nativeEval (S ≫ P) K (S ≫ G) ε' ε' n z x) := by
  rcases exists_native_cell (S ≫ P) K (S ≫ G) ε' ε' z with
    ⟨y, rfl⟩ | ⟨y, s, rfl⟩ | ⟨y, s, rfl⟩ | rfl | rfl
  · rw [preMinusConeMap, preConeMap, CutOpenFold.map_laurent, nativeEval_laurentInclusion]
    change ProjectionReference.FourEdge.coneEval X (A ≫ E) n
      (FourEdgeTree.laurentInclusion (A ≫ E) J L β
        (mapParameter (reanchor (projectionLaurentSeamAnchor f g c minus minus_projection)) y)) x
          = _
    rw [ProjectionReference.FourEdge.coneEval_laurent]
    change (mappingConeInclusion (comparison X)).map n
      (pointwisePrespectrumFamily L n
        (mapParameter (reanchor (projectionLaurentSeamAnchor f g c minus minus_projection)) y) x)
          = _
    rw [pointwise_reanchor]
    rfl
  · rw [preMinusConeMap, preConeMap, CutOpenFold.map_cylinder, nativeEval_cylinderMinus]
    change ProjectionReference.FourEdge.coneEval X (A ≫ E) n
      (FourEdgeTree.cylinder (A ≫ E) J L β
        (SSet.toTop.map (FourEdgeTree.minusVertex (A ≫ E) J)
          (mapParameter (reanchor (projectionSeamAnchor f g c)) y), s)) x = _
    rw [ProjectionReference.FourEdge.coneEval_cylinder, FourEdgeTree.nativeEval_minus]
    change Quillen.ReducedMappingCone.cylinder _ _ ((comparison X).map n)
      (s, (HomotopyPushout.inr (boundedFlatPerfectPrespectrumMap (projection.app X))
        (boundedFlatPerfectPrespectrumMap (projection.app X))).map n
          (pointwisePrespectrumFamily ((A ≫ E) ≫ J) n
            (mapParameter (reanchor (projectionSeamAnchor f g c)) y) x)) = _
    rw [pointwise_reanchor]
    rfl
  · rw [preMinusConeMap, preConeMap, CutOpenFold.map_cylinderPlus, nativeEval_cylinderPlus]
    change ProjectionReference.FourEdge.coneEval X (A ≫ E) n
      (FourEdgeTree.cylinder (A ≫ E) J L β
        (SSet.toTop.map (FourEdgeTree.minusVertex (A ≫ E) J)
          (mapParameter (reanchor (projectionSeamAnchor f g c)) y), s)) x = _
    rw [ProjectionReference.FourEdge.coneEval_cylinder, FourEdgeTree.nativeEval_minus]
    change Quillen.ReducedMappingCone.cylinder _ _ ((comparison X).map n)
      (s, (HomotopyPushout.inr (boundedFlatPerfectPrespectrumMap (projection.app X))
        (boundedFlatPerfectPrespectrumMap (projection.app X))).map n
          (pointwisePrespectrumFamily ((A ≫ E) ≫ J) n
            (mapParameter (reanchor (projectionSeamAnchor f g c)) y) x)) = _
    rw [pointwise_reanchor]
    rfl
  · rw [preMinusConeMap, preConeMap, CutOpenFold.map_tipMinus,
      ProjectionReference.FourEdge.coneEval_tip, nativeEval_tipMinus]
    exact ((InputIdentityNegative.inclusion X).map_point n).symm
  · rw [preMinusConeMap, preConeMap, CutOpenFold.map_tipPlus,
      ProjectionReference.FourEdge.coneEval_tip, nativeEval_tipPlus]
    exact ((InputIdentityNegative.inclusion X).map_point n).symm
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
