private theorem cancel_reanchor {V W : Model.{u}} {F G : V ⟶ W}
    (a : F.functor ≅ G.functor) (y : |nerve (Codiscrete (Node G))|) :
    mapParameter (reanchor a.symm) (mapParameter (reanchor a) y) = y := by
  obtain ⟨k,N,z,rfl⟩ := exists_parameter G y
  rw [mapParameter_parameter, mapParameter_parameter]
  apply congrArg (fun ns : Fin (k+1) → Node G ↦ parameter G ns z)
  funext i
  rcases N i with ⟨H,e⟩
  change (⟨H,a.symm ≪≫ (a ≪≫ e)⟩ : Node G) = ⟨H,e⟩
  rw [← Iso.trans_assoc, Iso.symm_self_id, Iso.refl_trans]

variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
local notation "E" => baseSource f g

/-- The original chosen cube, reanchored without changing any interior point. -/
def originalOverlapCube :=
  (mapParameter (reanchor (middleAnchor f g).symm)).comp
    (InputCompositionCarrierCube.carrierCube f g)

/-- The entire original cube top is the actual tree-composition attachment. -/
theorem originalOverlapCube_seam (r t : I) :
    originalOverlapCube f g ![r,1,t] =
      SSet.toTop.map (ProjectionReference.FourEdge.beta X E)
        (treeComposition f g (r,t)) := by
  change mapParameter _ (InputCompositionCarrierCube.carrierCube f g ![r,1,t]) = _
  rw [InputCompositionCarrierCube.cube_top, ← treeComposition_beta]
  exact cancel_reanchor (middleAnchor f g) _
/-- Every original cube value retains its literal native comparison value. -/
theorem originalOverlapCube_native (n : ℕ) (r s t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    pointwisePrespectrumFamily (ProjectionReference.reference X E) n
      (originalOverlapCube f g ![r,s,t]) x =
      (InputComposition.comparisonCylinder f g).level n (r,(s,(t,x))) := by
  change pointwisePrespectrumFamily _ n (mapParameter (reanchor (middleAnchor f g).symm)
    (InputCompositionCarrierCube.carrierCube f g ![r,s,t])) x = _
  exact (pointwise_reanchor (middleAnchor f g).symm n
    (InputCompositionCarrierCube.carrierCube f g ![r,s,t]) x).trans
      (pointwisePrespectrumFamily_eq_compact
        (InputCompositionCarrierCharts.source f g)
        (InputCompositionCarrierCube.carrierCube f g) n ![r,s,t] x)

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionSpanScalar
