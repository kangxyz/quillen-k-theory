/-- A separate-package consumer keeps the first selected comparison at its original lower half. -/
theorem first_comparison_cell (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    evaluate f g n x (surface f g (Quillen.NativePathCells.lower r, s)) =
      (InputCompositionNull.firstComparison f g).level n (r, (s, x)) := by
  rw [surface_lower]
  exact SeedSurfaceTransport.surface_native f g n r s x

end Consumer.GlobalRawBassInputNullJ1Carrier
