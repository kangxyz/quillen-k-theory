noncomputable section

open CategoryTheory AlgebraicGeometry Quillen.LoopPrespectrum

namespace Consumer.GlobalRawBassInputNullChosenCube
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
open InputNullCompositionChosenCube

/-- A downstream client varies structure time independently of all three comparison coordinates. -/
theorem independent_structure (n : ℕ) (q r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) (t : unitInterval) :
    chosenComparison f g (n+1)
      (q,(r,(s,(originalPrespectrum (boundedFlatPerfectWaldhausen Z)).structureMap n x t))) =
      (mappingCone (comparison X)).structureMap n
        (chosenComparison f g n (q,(r,(s,x)))) t :=
  chosenComparison_structure f g n q r s x t

end Consumer.GlobalRawBassInputNullChosenCube
