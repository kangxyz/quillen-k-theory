@[expose] public section
noncomputable section

universe u
namespace Consumer.GlobalSupportedEvaluation

/-- Both outside localization rows retain their actual original supported parameter sources. -/
def outsideRowEvaluations {U X : Scheme.{u}} (j : U ⟶ X) [IsOpenImmersion j]
    [CompactSpace U] [CompactSpace X] (T : X.Opens) :
    (strictSquareLoopedConeFunctor.obj (Source.parameterSquare (Rows.source j T)) ⟶
      step (Rows.source j T)) ×
    (strictSquareLoopedConeFunctor.obj (Source.parameterSquare (Rows.target j T)) ⟶
      step (Rows.target j T)) :=
  ⟨stepEvaluation _, stepEvaluation _⟩

/-- The source-support row's original seed compares under the same actual R1 evaluation. -/
def originalRowSeed {U X : Scheme.{u}} (j : U ⟶ X) [IsOpenImmersion j]
    [CompactSpace X] (T : X.Opens) :
    Homotopy (parameterSeed (Rows.source j T) ≫ stepEvaluation (Rows.source j T))
      (baseEvaluation (Rows.source j T) ≫ transition (Rows.source j T)) :=
  seedHomotopy _

/-- The middle support row preserves the actual supported Laurent evaluation at cone inclusion. -/
theorem middle_laurent_inclusion {X : Scheme.{u}} [CompactSpace X] (T : X.Opens) :
    mappingConeInclusion (StrictSquare.comparison (Source.sectionSquare (Rows.middle T))) ≫
        coneEvaluation (Rows.middle T) =
      evaluation (Geometry.laurent (Rows.middle T)) ≫
        mappingConeInclusion (comparison (Rows.middle T)) :=
  coneEvaluation_inclusion _

end Consumer.GlobalSupportedEvaluation
