module

public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassInputNullCompositionLaurentSides

/-! Client assembly of the complete scalar Laurent boundary, retaining the original paths. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
open CategoryTheory AlgebraicGeometry
open Quillen.Waldhausen.IteratedS
open Quillen.PerfectComplexes Quillen.PerfectComplexes.GlobalRawBass
open scoped unitInterval
universe u
namespace Consumer.GlobalRawBassInputNullCompositionLaurent
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- All four whole boundaries of the original Laurent square coexist in the fixed carrier. -/
theorem complete_boundary (r s : unitInterval) :
    InputNullCompositionLaurentCarrier.face f g (0, s) =
        InputIdentitySeedRectangle.laurentMovement (f ≫ g) false s ∧
    InputNullCompositionLaurentCarrier.face f g (1, s) =
        ((InputNullCompositionCompositeCarrier.firstPath f g).trans
          (InputNullCompositionCompositeCarrier.secondPath f g)) s ∧
    InputNullCompositionLaurentCarrier.face f g (r, 0) =
        InputNullCompositionPolynomialCarrier.sourceRail f g (r, 0) ∧
    InputNullCompositionLaurentCarrier.face f g (r, 1) =
        InputNullCompositionPolynomialCarrier.targetRail f g (r, 0) :=
  ⟨InputNullCompositionLaurentCarrier.face_zero f g s,
    InputNullCompositionLaurentCarrier.face_one f g s,
    InputNullCompositionLaurentCarrier.face_left f g r,
    InputNullCompositionLaurentCarrier.face_right f g r⟩

end Consumer.GlobalRawBassInputNullCompositionLaurent
