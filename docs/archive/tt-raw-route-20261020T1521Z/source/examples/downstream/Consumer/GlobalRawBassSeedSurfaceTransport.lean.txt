module
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassSeedSurfaceTransport

/-! An independent client extracts the original first-comparison target word. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
open CategoryTheory AlgebraicGeometry Quillen.LoopPrespectrum
open Quillen.Waldhausen.IteratedS
open Quillen.PerfectComplexes Quillen.PerfectComplexes.GlobalRawBass
open scoped Topology unitInterval
universe u
namespace Consumer.GlobalRawBassSeedSurfaceTransport
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- The literal target association evaluates to the maintained firstComparison target. -/
theorem original_middle_word (n : ℕ) (s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    (InputIdentityPositive.inclusion X).map n
      (CommonReferenceNerve.CutOpenCarrier.nativeEval
        (InputIdentitySeedCarrier.polyRef (f ≫ g)) (InputIdentitySeedCarrier.chart (X := X))
        (InputIdentitySeedCarrier.laurentRef (f ≫ g))
        (InputIdentitySeedCarrier.ordinary (f ≫ g)) (InputIdentitySeedCarrier.shifted (f ≫ g))
        n (((SeedSurfaceTransport.laurentMovement f g).trans
          (SeedSurfaceTransport.newNull f g)) s) x) =
      ((InputCompositionNull.firstLaurent f g).trans
        (InputCompositionNull.middleNull f g)).level n (s, x) := by
  rw [← SeedSurfaceTransport.comparison_target]
  exact (SeedSurfaceTransport.surface_native f g n 1 s x).trans
    (((InputCompositionNull.firstComparison f g).level n).apply_one (s, x))

end Consumer.GlobalRawBassSeedSurfaceTransport
