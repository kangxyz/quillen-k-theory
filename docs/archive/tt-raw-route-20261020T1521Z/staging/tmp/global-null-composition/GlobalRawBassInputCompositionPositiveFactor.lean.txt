/-!
# The exact positive support of original global raw composition

The prescribed paused positive chart and polynomial composition track supply a positive-cone
composition homotopy. Its strict image is the already chosen whole-cone homotopy at every
positive-cone point. The original terminal pause, cone reassociation, boundary collapse and
square descent are retained. No inverse chart or general overlap interior is reconstructed.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory AlgebraicGeometry Quillen.LoopPrespectrum

open InputIdentityPositive
variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)

/-- The original polynomial composition retains the fixed span law's terminal constant half. -/
def polynomialTrack := (InputCompositionPolynomial.polynomialComposition f g).trans
  (Quillen.LoopPrespectrum.Homotopy.refl (boundedFlatPerfectPrespectrumMap (polynomial.map g) ≫
    boundedFlatPerfectPrespectrumMap (polynomial.map f)))

/-- The two original positive squares have the exact square-cone composition association. -/
def compositeComparison := coneSquarePastedHomotopy (chart Z) (chart Y) (chart X)
  (boundedFlatPerfectPrespectrumMap (polynomial.map g))
  (boundedFlatPerfectPrespectrumMap (laurent.map g))
  (boundedFlatPerfectPrespectrumMap (polynomial.map f))
  (boundedFlatPerfectPrespectrumMap (laurent.map f))
  (plusSquareHomotopy g) (plusSquareHomotopy f)

/-- The whole positive face is the prescribed original geometric rectangle with its pause. -/
def comparisonSquare :
    (plusSquareHomotopy (f ≫ g)).Square (compositeComparison f g)
      (Homotopy.precomp (chart Z) (InputCompositionCharts.laurentComposition f g))
      ((polynomialTrack f g).postcomp (chart X)) := by
  refine ((InputCompositionTransformations.movingSquare plus f g).pauseRight).ofBoundaryLevels
    _ _ _ _ (fun _ ↦ rfl) (fun _ ↦ rfl) (fun _ ↦ rfl) ?_
  intro n
  rw [polynomialTrack, Homotopy.postcomp_trans]
  apply Homotopy.trans_level_congr <;> rfl

/-- The whole span law retains exactly the paused polynomial track on its positive chart. -/
theorem polynomialTrack_inl (n : ℕ) (r : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj Z))).space n) :
    (InputCompositionSpan.homotopy f g).level n
      (r, (HomotopyPushout.inl (boundedFlatPerfectPrespectrumMap (projection.app Z))
        (boundedFlatPerfectPrespectrumMap (projection.app Z))).map n x) =
      (HomotopyPushout.inl (boundedFlatPerfectPrespectrumMap (projection.app X))
        (boundedFlatPerfectPrespectrumMap (projection.app X))).map n
        ((polynomialTrack f g).level n (r, x)) := by
  rw [InputCompositionSpan.homotopy_inl]
  change ((Homotopy.levelHomotopy _ n).trans (Homotopy.levelHomotopy _ n)) (r, x) =
    (HomotopyPushout.inl _ _).map n
      (((Homotopy.levelHomotopy _ n).trans (Homotopy.levelHomotopy _ n)) (r, x))
  rw [ContinuousMap.HomotopyRel.trans_apply, ContinuousMap.HomotopyRel.trans_apply]
  split <;> rfl

/-- Positive composition keeps the entire original corrected moving-null square. -/
def nullSquare := coneSquareCompositionMovingNullSquare (chart Z) (chart Y) (chart X)
  (boundedFlatPerfectPrespectrumMap (polynomial.map g))
  (boundedFlatPerfectPrespectrumMap (laurent.map g))
  (boundedFlatPerfectPrespectrumMap (polynomial.map f))
  (boundedFlatPerfectPrespectrumMap (laurent.map f))
  (plusSquareHomotopy g) (plusSquareHomotopy f) (plusSquareHomotopy (f ≫ g))
  (polynomialTrack f g) (InputCompositionCharts.laurentComposition f g)
  (comparisonSquare f g)

/-- The actual positive input cone maps compose with their original paused chart choice. -/
def coneHomotopy : Homotopy (InputIdentityPositive.coneMap (f ≫ g))
    (InputIdentityPositive.coneMap g ≫ InputIdentityPositive.coneMap f) :=
  coneSquareCompositionHomotopyOfSquare (chart Z) (chart Y) (chart X)
    (boundedFlatPerfectPrespectrumMap (polynomial.map g))
    (boundedFlatPerfectPrespectrumMap (laurent.map g))
    (boundedFlatPerfectPrespectrumMap (polynomial.map f))
    (boundedFlatPerfectPrespectrumMap (laurent.map f))
    (plusSquareHomotopy g) (plusSquareHomotopy f) (plusSquareHomotopy (f ≫ g))
    (polynomialTrack f g) (InputCompositionCharts.laurentComposition f g)
    (comparisonSquare f g)

/-- The three original cone-null pieces keep their actual positive support through reassociation. -/
theorem nullCorrection_factor (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj Z))).space n) :
    (inclusion X).map n ((coneSquareCompositionNullComparison (chart Z) (chart Y) (chart X)
      (boundedFlatPerfectPrespectrumMap (polynomial.map g))
      (boundedFlatPerfectPrespectrumMap (laurent.map g))
      (boundedFlatPerfectPrespectrumMap (polynomial.map f))
      (boundedFlatPerfectPrespectrumMap (laurent.map f))
      (plusSquareHomotopy g) (plusSquareHomotopy f)).level n (r, (s, x))) =
    (coneSquareCompositionNullComparison (comparison Z) (comparison Y) (comparison X)
      (spanMap g) (boundedFlatPerfectPrespectrumMap (laurent.map g))
      (spanMap f) (boundedFlatPerfectPrespectrumMap (laurent.map f))
      (inputComparisonHomotopy g) (inputComparisonHomotopy f)).level n
        (r, (s, (HomotopyPushout.inl (boundedFlatPerfectPrespectrumMap (projection.app Z))
          (boundedFlatPerfectPrespectrumMap (projection.app Z))).map n x)) := by
  apply coneSquareCompositionNullComparison_mapped_points
  all_goals intro t; rfl

/-- Every value of the fixed whole composition null square lies in its original positive support. -/
theorem nullSquare_factor (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj Z))).space n) :
    (inclusion X).map n ((nullSquare f g).level n (r, (s, x))) =
      (InputComposition.comparisonConeNullSquare f g).level n
        (r, (s, (HomotopyPushout.inl (boundedFlatPerfectPrespectrumMap (projection.app Z))
          (boundedFlatPerfectPrespectrumMap (projection.app Z))).map n x)) := by
  unfold nullSquare InputComposition.comparisonConeNullSquare
    coneSquareCompositionMovingNullSquare
  apply Homotopy.RelHomotopy.toSquare_mapped_points
  intro r' s'
  apply Homotopy.RelHomotopy.trans_mapped_points
  · intro r'' s''
    apply Homotopy.Square.toBoundary_mapped_points
    intro r''' s'''
    unfold coneSquareNullMovingSquare
    apply Homotopy.Square.pastePaths_mapped_points
    · intro t w
      rfl
    · intro t w
      change ReducedMappingCone.cylinder _ _ ((comparison X).map n)
        (w, (HomotopyPushout.inl _ _).map n ((polynomialTrack f g).level n (t, x))) =
          ReducedMappingCone.cylinder _ _ ((comparison X).map n)
            (w, (InputCompositionSpan.homotopy f g).level n
              (t, (HomotopyPushout.inl _ _).map n x))
      rw [polynomialTrack_inl]
  · intro r'' s''
    apply Homotopy.RelHomotopy.hcomp_mapped_points
    · intro r''' s'''
      apply Homotopy.RelHomotopy.hcomp_mapped_points
      · intro _ _
        rfl
      · intro t w
        exact nullCorrection_factor f g n t w x
    · intro _ _
      rfl

/-- The fixed global homotopy agrees with the positive construction on every positive-cone point. -/
theorem coneHomotopy_factor (n : ℕ) (r : unitInterval)
    (x : (mappingCone (chart Z)).space n) :
    (inclusion X).map n ((coneHomotopy f g).level n (r, x)) =
      (InputComposition.coneHomotopy f g).level n (r, (inclusion Z).map n x) := by
  have h : (coneHomotopy f g).at r ≫ inclusion X =
      inclusion Z ≫ (InputComposition.coneHomotopy f g).at r := by
    apply mappingCone_hom_ext
    · apply Hom.ext
      funext k
      ext y
      rfl
    · intro k s y
      exact nullSquare_factor f g k r s y
  exact congrArg (fun k ↦ k.map n x) h

/-- The whole source rail already has an actual positive-cone representative. -/
def sourceRail (n : ℕ) :
    C(unitInterval × (unitInterval ×
      (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n),
      (mappingCone (chart X)).space n) :=
  ⟨fun z ↦ (coneHomotopy f g).level n
    (z.1, (InputIdentityPositive.sourceNull Z).level n z.2), by fun_prop⟩

/-- The positive representative recovers the original full source rail at both unchanged clocks. -/
theorem sourceRail_factor (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    (inclusion X).map n (sourceRail f g n (r, (s, x))) =
      (InputComposition.coneHomotopy f g).level n
        (r, (GlobalRawBass.sourceNull Z).level n (s, x)) := by
  change (inclusion X).map n ((coneHomotopy f g).level n
    (r, (InputIdentityPositive.sourceNull Z).level n (s, x))) = _
  rw [coneHomotopy_factor]
  exact congrArg (fun y ↦ (InputComposition.coneHomotopy f g).level n (r, y))
    (DFunLike.congr_fun (InputIdentityPositive.sourceNull_factor Z n) (s, x))

/-- The supported whole rail fixes the basepoint for every pair of original times. -/
theorem sourceRail_point (n : ℕ) (r s : unitInterval) :
    sourceRail f g n (r, (s,
      (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).point n)) =
        (mappingCone (chart X)).point n := by
  change (coneHomotopy f g).level n
    (r, (InputIdentityPositive.sourceNull Z).level n (s, _)) = _
  exact (congrArg (fun y ↦ (coneHomotopy f g).level n (r, y))
    ((InputIdentityPositive.sourceNull Z).at_point n s)).trans
      ((coneHomotopy f g).at_point n r)

/-- Its prespectrum structure time remains independent of the two rail coordinates. -/
theorem sourceRail_structure (n : ℕ) (r s : unitInterval)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n)
    (t : unitInterval) :
    sourceRail f g (n + 1) (r, (s,
      (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).structureMap n x t)) =
        (mappingCone (chart X)).structureMap n (sourceRail f g n (r, (s, x))) t := by
  change (coneHomotopy f g).level (n + 1)
    (r, (InputIdentityPositive.sourceNull Z).level (n + 1) (s, _)) = _
  exact (congrArg (fun y ↦ (coneHomotopy f g).level (n + 1) (r, y))
    ((InputIdentityPositive.sourceNull Z).comm n s x t)).trans
      ((coneHomotopy f g).comm n r ((InputIdentityPositive.sourceNull Z).level n (s, x)) t)
