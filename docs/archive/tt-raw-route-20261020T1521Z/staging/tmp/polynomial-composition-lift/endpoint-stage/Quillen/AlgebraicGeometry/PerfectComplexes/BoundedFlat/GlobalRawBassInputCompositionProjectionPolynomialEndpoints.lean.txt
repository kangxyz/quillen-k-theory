/-- Tree Composition left for the exact original reference and parameter data. -/
theorem treeComposition_left (r : unitInterval) :
    InputCompositionSpanScalar.treeComposition f g (r, 0) =
      SSet.toTop.map (FourEdgeTree.plusVertex E (ProjectionReference.projectionMap X))
        (mapParameter (polynomialNode f g)
          (InputCompositionPolynomialScalar.pausedPolynomial f g r)) := by
  rw [InputCompositionSpanScalar.treeComposition, FourEdgeTree.spanComposition_left]
  rcases cover r with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [path_lower, InputCompositionPolynomialScalar.pausedPolynomial, path_lower]
    exact congrArg (SSet.toTop.map (FourEdgeTree.plusVertex E
      (ProjectionReference.projectionMap X))) (polynomialSquare_left f g t)
  · rw [path_upper, InputCompositionPolynomialScalar.pausedPolynomial, path_upper]
    have ht := polynomialSquare_left f g 1
    rw [Path.target] at ht
    exact congrArg (SSet.toTop.map (FourEdgeTree.plusVertex E
      (ProjectionReference.projectionMap X))) ht

/-- Tree Composition right for the exact original reference and parameter data. -/
theorem treeComposition_right (r : unitInterval) :
    InputCompositionSpanScalar.treeComposition f g (r, 1) =
      SSet.toTop.map (FourEdgeTree.minusVertex E (ProjectionReference.projectionMap X))
        (mapParameter (polynomialNode f g)
          (InputCompositionPolynomialScalar.pausedPolynomial f g r)) := by
  rw [InputCompositionSpanScalar.treeComposition, FourEdgeTree.spanComposition_right]
  rcases cover r with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [path_lower, InputCompositionPolynomialScalar.pausedPolynomial, path_lower]
    exact congrArg (SSet.toTop.map (FourEdgeTree.minusVertex E
      (ProjectionReference.projectionMap X))) (polynomialSquare_left f g t)
  · rw [path_upper, InputCompositionPolynomialScalar.pausedPolynomial, path_upper]
    have ht := polynomialSquare_left f g 1
    rw [Path.target] at ht
    exact congrArg (SSet.toTop.map (FourEdgeTree.minusVertex E
      (ProjectionReference.projectionMap X))) ht
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionProjectionAttachment
