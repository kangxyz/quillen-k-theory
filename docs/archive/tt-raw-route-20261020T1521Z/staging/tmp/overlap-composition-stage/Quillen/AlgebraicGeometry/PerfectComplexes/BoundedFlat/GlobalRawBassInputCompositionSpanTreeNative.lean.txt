variable {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
local notation "E" => baseSource f g
local notation "J" => ProjectionReference.projectionMap X

/-- Every original moving-stage value recovers the selected native span square. -/
theorem treeMoving_native (n : ℕ) (r t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    FourEdgeTree.nativeEval E J n (treeMoving f g (r,t)) x =
      (InputCompositionSpan.movingSquare f g).level n (r,(t,x)) := by
  exact FourEdgeTree.movingSquare_native E J (polynomialSquare f g)
    (basePath f g) (polynomialSquare_attachment f g)
    (boundedFlatPerfectPrespectrumMap (projection.app Z))
    (polynomialSquareHomotopy (f ≫ g)) (InputCompositionSpan.pastedSquare f g)
    (InputCompositionPolynomial.sourceComposition f g)
    (InputCompositionPolynomial.polynomialComposition f g) _ n x
    (fun r t ↦ polynomialSquare_native f g n r t x)
    (fun r ↦ basePath_native f g n r x) r t

/-- The complete original five-piece reassociation retains both native parameters. -/
theorem treeReassociation_native (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    FourEdgeTree.nativeEval E J n (FourEdgeTree.spanReassociation E J (basePath f g 1)
      (firstPolynomial f g) (lastPolynomial f g) (r,s)) x =
      (InputCompositionTopAssoc.reassociationNative f g).level n (r,(s,x)) := by
  let a := boundedFlatPerfectPrespectrumMap g
  let c := boundedFlatPerfectPrespectrumMap f
  let d := boundedFlatPerfectPrespectrumMap (polynomial.map f)
  let h := boundedFlatPerfectPrespectrumMap (projection.app X)
  let j := boundedFlatPerfectPrespectrumMap (projection.app Y)
  let H := polynomialSquareHomotopy g
  let K := polynomialSquareHomotopy f
  let iL := HomotopyPushout.inl h h
  let iR := HomotopyPushout.inr h h
  let P := H.postcomp (d ≫ iL)
  let Q : Homotopy ((a ≫ j) ≫ (d ≫ iL)) ((a ≫ c) ≫ (h ≫ iL)) :=
    (Homotopy.precomp a (K.postcomp iL)).cast
      (by simp only [d, h, j, Category.assoc]) (by simp only [c, h, Category.assoc])
  let T := Homotopy.precomp (a ≫ c) (HomotopyPushout.glue h h)
  let U : Homotopy ((a ≫ c) ≫ (h ≫ iR)) ((a ≫ j) ≫ (d ≫ iR)) :=
    (Homotopy.precomp a (K.symm.postcomp iR)).cast
      (by simp only [c, h, Category.assoc]) (by simp only [d, h, j, Category.assoc])
  let V := H.symm.postcomp (d ≫ iR)
  have hn (t v : I) :
      Path.Homotopy.spanReassociation (P.path n x) (Q.path n x) (T.path n x)
        (U.path n x) (V.path n x) (t,v) =
      (InputCompositionTopAssoc.reassociationNative f g).level n (t,(v,x)) := by
    exact Quillen.NativeSpanReassociation.word n x id
      (P.path n x) (Q.path n x) (T.path n x) (U.path n x) (V.path n x)
      P Q T U V (fun _ ↦ rfl) (fun _ ↦ rfl) (fun _ ↦ rfl)
      (fun _ ↦ rfl) (fun _ ↦ rfl) t v
  let ev := fun z ↦ FourEdgeTree.nativeEval E J n z x
  have ht (v : I) :
      ev (FourEdgeTree.spanReassociation E J (basePath f g 1)
        (firstPolynomial f g) (lastPolynomial f g) (1,v)) =
      (((P.path n x).trans (Q.path n x)).trans
        ((T.path n x).trans ((U.path n x).trans (V.path n x)))) v := by
    have hj := FourEdgeTree.spanComposition_join E J (basePath f g 1)
      (firstPolynomial f g) (lastPolynomial f g) (polynomialSquare f g)
      (basePath f g) (polynomialSquare_attachment f g) rfl (polynomialTop_pasted f g) v
    have he := (congrArg ev hj).symm.trans (treeMoving_native f g n 1 v x)
    have htop := hn 1 v
    rw [ContinuousMap.HomotopyWith.apply_one] at htop
    exact he.trans ((((InputCompositionSpan.movingSquare f g).at_one n v x).trans
      (((InputCompositionTopAssoc.reassociationNative f g).level n).apply_one (v,x)).symm).trans
        htop.symm)
  have ht' (v : I) := ht v
  simp only [FourEdgeTree.spanReassociation, ContinuousMap.HomotopyWith.apply_one,
    Path.coe_toContinuousMap] at ht'
  exact (Path.Evaluation.spanReassociation_of_target_function ev (ContinuousMap.id _)
    _ _ _ _ _ (P.path n x) (Q.path n x) (T.path n x) (U.path n x) (V.path n x)
    ht' r s).trans (hn r s)

/-- Every whole tree-composition value recovers the original span homotopy. -/
theorem treeComposition_native (n : ℕ) (r t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen Z)).space n) :
    FourEdgeTree.nativeEval E J n (treeComposition f g (r,t)) x =
      (InputCompositionSpan.homotopy f g).level n (r, HomotopyPushout.cylinder
        (boundedFlatPerfectPrespectrumMap (projection.app Z))
        (boundedFlatPerfectPrespectrumMap (projection.app Z)) n (t,x)) := by
  rcases cover r with ⟨s,rfl⟩ | ⟨s,rfl⟩
  · rw [InputCompositionSpan.homotopy_cylinder_lower]
    change FourEdgeTree.nativeEval E J n
      (FourEdgeTree.spanComposition _ _ _ _ _ _ _ _ _ _ _) x = _
    rw [FourEdgeTree.spanComposition_lower]
    exact treeMoving_native f g n s t x
  · rw [InputCompositionSpan.homotopy_cylinder_upper]
    change FourEdgeTree.nativeEval E J n
      (FourEdgeTree.spanComposition _ _ _ _ _ _ _ _ _ _ _) x = _
    rw [FourEdgeTree.spanComposition_upper]
    exact treeReassociation_native f g n (unitInterval.symm s) t x
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionSpanScalar
