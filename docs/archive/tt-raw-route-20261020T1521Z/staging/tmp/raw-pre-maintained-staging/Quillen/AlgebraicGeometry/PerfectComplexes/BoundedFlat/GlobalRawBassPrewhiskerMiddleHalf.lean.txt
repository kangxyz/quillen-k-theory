open Quillen.NativePathCells
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "J" => ProjectionReference.projectionMap X
local notation "F" => ProjectionReference.reference X (A ≫ E)
local notation "β" => ProjectionReference.FourEdge.beta X (A ≫ E)

/-- The actual original middle branch glues at half height, retaining both independent clocks. -/
theorem middle_halfHeight (r t : I) :
    FourEdgeTree.laurentInclusion (A ≫ E) J F β
      (anchoredLowerLaurent f g c ((r, 1), upper (lower t))) =
      selectedOverlap f g c ((r, 0), t) := by
  have h := congrArg (fun p ↦ p t)
    ((InputCompositionOverlapScalar.correctedSquare f g).left r)
  change (InputCompositionOverlapScalar.correctedFamily f g) ((r, 0), t) =
    FourEdgeTree.laurentInclusion E J (ProjectionReference.reference X E)
      (ProjectionReference.FourEdge.beta X E)
      (InputCompositionSpanScalar.originalOverlapCube f g ![r, 0, t]) at h
  have hu := (congrArg (ProjectionReference.FourEdge.Precomp.cone X A E) h).trans
    (ProjectionReference.FourEdge.Precomp.cone_laurent X A E
      (InputCompositionSpanScalar.originalOverlapCube f g ![r, 0, t]))
  have hc := congrArg (fun y ↦ FourEdgeTree.laurentInclusion (A ≫ E) J F β
    (mapParameter (precompNode A)
      (mapParameter (CommonReferenceProduct.reanchor
        (InputCompositionTopPieces.middleAnchor f g).symm) y)))
    (InputCompositionCarrierCube.cube_bottom f g r t)
  exact (congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J F β)
    (middle_parameter f g c r t)).trans (hc.symm.trans hu.symm)

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
