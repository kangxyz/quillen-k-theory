private theorem parameter_square
    {U₁ V₁ U₂ V₂ U₃ V₃ U₄ V₄ : Model.{u}}
    {F₁ : U₁ ⟶ V₁} {F₂ : U₂ ⟶ V₂} {F₃ : U₃ ⟶ V₃} {F₄ : U₄ ⟶ V₄}
    (a : Node F₁ → Node F₂) (b : Node F₂ → Node F₄)
    (c : Node F₁ → Node F₃) (d : Node F₃ → Node F₄)
    (h : ∀ N, b (a N) = d (c N)) (y : |parameterSSet F₁|) :
    mapParameter b (mapParameter a y) = mapParameter d (mapParameter c y) := by
  obtain ⟨p, N, w, rfl⟩ := exists_parameter F₁ y
  rw [mapParameter_parameter, mapParameter_parameter, mapParameter_parameter,
    mapParameter_parameter]
  exact congrArg (fun ns ↦ parameter F₄ ns w) (funext fun i ↦ h (N i))

/-- The reanchored Laurent source is exactly the original overlap node, with its full anchor. -/
theorem old_laurent_node (N : Node P) :
    reanchor (overlapLaurentAnchor f g k hk) (precompNode D N) =
      InputCompositionProjectionAttachment.laurentNode f g k hk N := by
  simp only [overlapLaurentAnchor, InputCompositionProjectionAttachment.laurentNode,
    InputCompositionPolynomialOverlap.overlapNode, reanchor, precompNode, Iso.trans_assoc]

/-- The entire Laurent parameter at node four agrees with the actual fixed-source overlap map. -/
theorem terminal_laurent_parameter (y : |parameterSSet P|) :
    mapParameter (reanchor (projectionLaurentSeamAnchor f g c k hk))
      (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4)) =
    mapParameter (precompNode A)
      (mapParameter (InputCompositionProjectionAttachment.laurentNode f g k hk) y) := by
  have hv := FiniteParameter.parameter_vertex P (S ≫ P)
    (fun N i ↦ FinitePrecomp.node S P (M i) N) 4 y
  exact (congrArg (mapParameter (reanchor (projectionLaurentSeamAnchor f g c k hk))) hv).trans
    (parameter_square (fun N ↦ FinitePrecomp.node S P (M 4) N)
      (reanchor (projectionLaurentSeamAnchor f g c k hk))
      (InputCompositionProjectionAttachment.laurentNode f g k hk) (precompNode A)
      (fun N ↦ (terminal_laurent_fixed_node f g c k hk N).trans
        (congrArg (precompNode A) (old_laurent_node f g k hk N))) y)
