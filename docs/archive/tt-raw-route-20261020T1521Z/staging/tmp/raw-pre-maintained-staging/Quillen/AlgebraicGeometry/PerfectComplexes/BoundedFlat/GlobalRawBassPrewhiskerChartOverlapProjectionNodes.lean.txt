  (hk : ∀ V, k.app V ≫ projection.app V = laurentProjection.app V)
local notation "E" => InputCompositionTopPieces.baseSource f g

/-- Attachment agrees on every full source-square polynomial node, not only its terminal vertex. -/
theorem projectionSeam_attachment_node (N : Node (S ≫ P)) :
    reanchor (projectionLaurentSeamAnchor f g c k hk)
      (attachPostcompNode (S ≫ P) K (S ≫ G) (CutOpenPrecomp.anchor S P K G ε) N) =
      ProjectionReference.chartNode X k hk (A ≫ E)
        (reanchor (projectionSeamAnchor f g c) N) := by
  apply Sigma.ext
  · rfl
  apply heq_of_eq
  change projectionLaurentSeamAnchor f g c k hk ≪≫
      (CutOpenPrecomp.anchor S P K G ε ≪≫ isoWhiskerRight N.2 (K).functor) =
    ProjectionReference.chartAnchor X k hk (A ≫ E) ≪≫
      isoWhiskerRight (projectionSeamAnchor f g c ≪≫ N.2) (K).functor
  rw [isoWhiskerRight_trans, ← Iso.trans_assoc, projectionSeam_attachment, Iso.trans_assoc]
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
