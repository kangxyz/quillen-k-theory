open CommonReferenceNerve
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

/-- Every Laurent native input follows the complete original normalization binary path. -/
theorem lowerLaurentPath_native (n : ℕ) (r : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (laurent.obj Z))).space n) :
    pointwisePrespectrumFamily (lowerLaurentBinary f g) n (lowerLaurentPath f g r) x =
      (InputCompositionCharts.laurentComposition f g).level n (r, x) := by
  change pointwisePrespectrumFamily _ n (parameter _ (lowerLaurentNodes f g) _) x = _
  rw [pointwisePrespectrumFamily_parameter, prespectrumFamily_eq]
  exact BinaryNormalizationSimplex.native_path (laurent.map f) (laurent.map g) n r x

/-- The original thirty-one-node input comparison is retained before Laurent composition. -/
theorem lowerLaurent_native (n : ℕ) (r s t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen W)).space n) :
    pointwisePrespectrumFamily (lowerLaurentReference f g c) n
      (lowerLaurent f g c ((r, s), t)) x =
      (InputCompositionCharts.laurentComposition f g).level n
        (r, (inputComparisonHomotopy c).level n
          (s, Quillen.LoopPrespectrum.HomotopyPushout.cylinder
            (boundedFlatPerfectPrespectrumMap (projection.app W))
            (boundedFlatPerfectPrespectrumMap (projection.app W)) n (t, x))) := by
  change pointwisePrespectrumFamily _ n
    (FinitePrecomp.parameter (InputDescSimplex.reference c) (InputDescSimplex.nodes c)
      (lowerLaurentBinary f g)
      (lowerLaurentPath f g r, BassH0Diagram.parameter (s, t))) x = _
  rw [FinitePrecomp.parameter_native, lowerLaurentPath_native, prespectrumFamily_eq]
  exact congrArg (fun y ↦ (InputCompositionCharts.laurentComposition f g).level n (r, y))
    (InputDescSimplex.cylinder_native c n s t x)

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
