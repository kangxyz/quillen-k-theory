open CommonReferenceNerve
attribute [local irreducible] boundedFlatNormalizedPullback
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

/-- The source input-comparison projection is moved only by actual Laurent naturality. -/
theorem lowerLaurent_total :
    (laurent.map f ≫ laurent.map g) ≫ (laurent.map c ≫ laurentProjection.app W) =
      (laurent.map (f ≫ g) ≫ laurentProjection.app Z) ≫ c := by
  rw [← laurent.map_comp f g, laurentProjection.naturality c, Category.assoc]
  rfl

/-- The exact normalized composite reference joins the already chosen whole overlap reference. -/
def lowerLaurentAnchor :
    (ProjectionReference.reference X
      (BassH0Merge.normalizedMap c ≫ InputCompositionTopPieces.baseSource f g)).functor ≅
      (lowerLaurentReference f g c).functor :=
  Functor.associator (BassH0Merge.normalizedMap c).functor
      (InputCompositionTopPieces.baseSource f g).functor (InputIdentityProjection.source
        X).functor ≪≫
    isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
      (InputCompositionTopPieces.middleAnchor f g).symm ≪≫
    (BinaryNormalizationSimplex.total
      (laurent.map (f ≫ g) ≫ laurentProjection.app Z) c).symm ≪≫
    isoWhiskerLeft AsSmall.down (isoWhiskerRight
      (eqToIso (congrArg boundedFlatNormalizedPullback (lowerLaurent_total f g c).symm))
      AsSmall.up) ≪≫
    BinaryNormalizationSimplex.total (laurent.map f ≫ laurent.map g)
      (laurent.map c ≫ laurentProjection.app W)

/-- The full lower-half Laurent surface now occupies the chosen overlap reference. -/
def anchoredLowerLaurent : C((I × I) × I,
    |CutOpenCarrier.parameterSSet (ProjectionReference.reference X
      (BassH0Merge.normalizedMap c ≫ InputCompositionTopPieces.baseSource f g))|) :=
  (mapParameter (CommonReferenceProduct.reanchor (lowerLaurentAnchor f g c))).comp
    (lowerLaurent f g c)

/-- Reanchoring retains the original source-map first half at every independent native coordinate.
-/
theorem anchoredLowerLaurent_native (n : ℕ) (r s t : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen W)).space n) :
    pointwisePrespectrumFamily (ProjectionReference.reference X
      (BassH0Merge.normalizedMap c ≫ InputCompositionTopPieces.baseSource f g)) n
      (anchoredLowerLaurent f g c ((r, s), t)) x =
      (InputCompositionCharts.laurentComposition f g).level n
        (r, (inputComparisonHomotopy c).level n
          (s, Quillen.LoopPrespectrum.HomotopyPushout.cylinder
            (boundedFlatPerfectPrespectrumMap (projection.app W))
            (boundedFlatPerfectPrespectrumMap (projection.app W)) n (t, x))) :=
  (pointwise_reanchor (lowerLaurentAnchor f g c) n
    (lowerLaurent f g c ((r, s), t)) x).trans (lowerLaurent_native f g c n r s t x)

end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
