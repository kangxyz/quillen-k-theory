The whole lower positive sector meets the actual overlap source edge.
-/
@[expose] public section
noncomputable section

namespace Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
open CommonReferenceNerve
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "J" => ProjectionReference.projectionMap X
local notation "L" => ProjectionReference.reference X (A ≫ E)
local notation "β" => ProjectionReference.FourEdge.beta X (A ≫ E)

/-- The whole lower positive sector meets the actual overlap source edge. -/
theorem source_lower_plus (r s : I) :
    outerPlusConeMap f g c (polynomialLower f g c plus (r, s)) =
      overlapLower f g c ((r, s), 0) := by
  simp only [polynomialLower, ContinuousMap.comp_apply, outerPlusConeMap, outerConeMap,
    CutOpenFold.map_laurent]
  exact congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L β)
    (lowerChart_plus_projection f g c r s)

/-- The whole lower negative sector meets the actual overlap target edge. -/
theorem source_lower_minus (r s : I) :
    outerMinusConeMap f g c (polynomialLower f g c minus (r, s)) =
      overlapLower f g c ((r, s), 1) := by
  simp only [polynomialLower, ContinuousMap.comp_apply, outerMinusConeMap, outerConeMap,
    CutOpenFold.map_laurent]
  exact congrArg (FourEdgeTree.laurentInclusion (A ≫ E) J L β)
    (lowerChart_minus_projection f g c r s)

/-- The complete original positive source attachment holds at every independent time. -/
theorem source_plus (r s : I) :
    outerPlusConeMap f g c (polynomialPrewhisker f g c plus (r, s)) =
      overlapPrewhisker f g c ((r, s), 0) := by
  rcases cover s with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [polynomialPrewhisker_lower, overlapPrewhisker_lower]
    exact source_lower_plus f g c r t
  · rw [polynomialPrewhisker_upper, overlapPrewhisker_upper]
    exact source_upper_plus f g c r t

/-- The complete original negative source attachment holds at every independent time. -/
theorem source_minus (r s : I) :
    outerMinusConeMap f g c (polynomialPrewhisker f g c minus (r, s)) =
      overlapPrewhisker f g c ((r, s), 1) := by
  rcases cover s with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [polynomialPrewhisker_lower, overlapPrewhisker_lower]
    exact source_lower_minus f g c r t
  · rw [polynomialPrewhisker_upper, overlapPrewhisker_upper]
    exact source_upper_minus f g c r t
end Quillen.PerfectComplexes.GlobalRawBass.InputCompositionPrewhisker
