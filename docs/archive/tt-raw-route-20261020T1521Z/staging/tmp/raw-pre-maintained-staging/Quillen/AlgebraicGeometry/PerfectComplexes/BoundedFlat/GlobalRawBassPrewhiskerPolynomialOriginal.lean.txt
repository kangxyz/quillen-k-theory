/-- The complete genuine plus polynomial sector recovers the original whole raw prewhiskering. -/
theorem polynomialPrewhisker_plus_original (n : ℕ) (r s : I)
    (x : (originalPrespectrum (boundedFlatPerfectWaldhausen (polynomial.obj W))).space n) :
    (InputIdentityPositive.inclusion X).map n
      (CutOpenCarrier.nativeEval
        (BassH0Merge.normalizedMap (polynomial.map c) ≫
          InputCompositionPolynomialScalar.polynomialReference f g)
        (InputCompositionPolynomialScalar.chart (X := X) plus)
        (BassH0Merge.normalizedMap (polynomial.map c) ≫
          InputCompositionPolynomialScalar.laurentReference f g plus)
        (CutOpenPrecomp.anchor _ _ _ _ (InputCompositionPolynomialScalar.anchor f g plus))
        (CutOpenPrecomp.anchor _ _ _ _ (InputCompositionPolynomialScalar.anchor f g plus)) n
        (polynomialPrewhisker f g c plus (r, s)) x) =
    (InputComposition.coneHomotopy f g).level n
      (r, (GlobalRawBass.coneMap c).map n
        (cylinder W n (s, (HomotopyPushout.inl
          (boundedFlatPerfectPrespectrumMap (projection.app W))
          (boundedFlatPerfectPrespectrumMap (projection.app W))).map n x))) := by
  rcases cover s with ⟨t, rfl⟩ | ⟨t, rfl⟩
  · rw [polynomialPrewhisker_lower, polynomialLower_native, composition_lower,
      inputComparisonHomotopy_inl]
    rfl
  · rw [polynomialPrewhisker_upper, polynomialUpper_native, composition_upper]
    exact (InputCompositionPositive.nullSquare_factor f g n r t
      ((boundedFlatPerfectPrespectrumMap (polynomial.map c)).map n x)).trans
      (congrArg (fun y ↦ (InputComposition.comparisonConeNullSquare f g).level n (r, (t, y)))
        (congrArg (fun h ↦ h.map n x) (inl_spanMap c)).symm)
