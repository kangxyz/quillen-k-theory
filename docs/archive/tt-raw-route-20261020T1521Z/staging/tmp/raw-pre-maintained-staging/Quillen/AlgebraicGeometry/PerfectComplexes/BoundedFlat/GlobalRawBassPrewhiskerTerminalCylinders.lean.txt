local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "S" => NativeSquareFiveSimplex.source (polynomial.map c) (projection.app W)
local notation "M" => NativeSquareFiveSimplex.nodes (polynomial.map c) (projection.app W)

local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "J" => ProjectionReference.projectionMap X

local notation "L'" => ProjectionReference.reference X (A ≫ E)
local notation "β" => ProjectionReference.FourEdge.beta X E
local notation "β'" => ProjectionReference.FourEdge.beta X (A ≫ E)

/-- Every positive polynomial cylinder agrees at the true terminal source-square seam. -/
theorem terminal_plus_cylinder (y : |parameterSSet P|) (s : I) :
    preCylinderMap f g c (FourEdgeTree.plusVertex (A ≫ E) J)
      (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4), s) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (InputCompositionProjectionAttachment.cylinderMap f g (FourEdgeTree.plusVertex E J) (y, s))
        := by
  change FourEdgeTree.cylinder (A ≫ E) J L' β'
      (SSet.toTop.map (FourEdgeTree.plusVertex (A ≫ E) J)
        (mapParameter (reanchor (projectionSeamAnchor f g c))
          (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4))), s) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (FourEdgeTree.cylinder E J L β
        (SSet.toTop.map (FourEdgeTree.plusVertex E J)
          (mapParameter (InputCompositionProjectionAttachment.polynomialNode f g) y), s))
  rw [ProjectionReference.FourEdge.Precomp.cone_cylinder,
    ProjectionReference.FourEdge.Precomp.tree_plus]
  exact congrArg (fun z ↦ FourEdgeTree.cylinder (A ≫ E) J L' β'
    (SSet.toTop.map (FourEdgeTree.plusVertex (A ≫ E) J) z, s))
      (terminal_polynomial_parameter f g c y)

/-- The negative cylinder keeps its distinct original vertex and the same full-height seam. -/
theorem terminal_minus_cylinder (y : |parameterSSet P|) (s : I) :
    preCylinderMap f g c (FourEdgeTree.minusVertex (A ≫ E) J)
      (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4), s) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (InputCompositionProjectionAttachment.cylinderMap f g (FourEdgeTree.minusVertex E J) (y, s))
        := by
  change FourEdgeTree.cylinder (A ≫ E) J L' β'
      (SSet.toTop.map (FourEdgeTree.minusVertex (A ≫ E) J)
        (mapParameter (reanchor (projectionSeamAnchor f g c))
          (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4))), s) =
    ProjectionReference.FourEdge.Precomp.cone X A E
      (FourEdgeTree.cylinder E J L β
        (SSet.toTop.map (FourEdgeTree.minusVertex E J)
          (mapParameter (InputCompositionProjectionAttachment.polynomialNode f g) y), s))
  rw [ProjectionReference.FourEdge.Precomp.cone_cylinder,
    ProjectionReference.FourEdge.Precomp.tree_minus]
  exact congrArg (fun z ↦ FourEdgeTree.cylinder (A ≫ E) J L' β'
    (SSet.toTop.map (FourEdgeTree.minusVertex (A ≫ E) J) z, s))
      (terminal_polynomial_parameter f g c y)
