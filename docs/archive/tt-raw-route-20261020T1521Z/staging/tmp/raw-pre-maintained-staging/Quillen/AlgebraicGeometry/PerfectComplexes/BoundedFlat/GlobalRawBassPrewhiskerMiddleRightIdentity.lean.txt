theorem middleRightIdentity : Functor.associator (BassH0Merge.normalizedMap c).functor
      (InputCompositionTopPieces.baseSource f g).functor
      (InputIdentityProjection.source X).functor ≪≫
      isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
        ((InputCompositionTopPieces.middleAnchor f g).symm ≪≫
          InputCompositionCarrierBottom.sourceIso f g) =
      isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
        (InputCompositionTopPieces.middleAnchor f g).symm ≪≫
      isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
        (InputCompositionCarrierBottom.sourceIso f g) :=
  (Iso.refl_trans (isoWhiskerLeft (BassH0Merge.normalizedMap c).functor
    ((InputCompositionTopPieces.middleAnchor f g).symm ≪≫
      InputCompositionCarrierBottom.sourceIso f g))).trans
    (isoWhiskerLeft_trans (BassH0Merge.normalizedMap c).functor
      (InputCompositionTopPieces.middleAnchor f g).symm
      (InputCompositionCarrierBottom.sourceIso f g))
