/-!
A specified full source node retains its anchor under fixed input precomposition.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory CategoryTheory.Functor
universe u
namespace Quillen.Waldhausen.IteratedS.CommonReferenceProduct
open CommonReferenceNerve
variable {U V W T : Model.{u}} (R : U ⟶ V) (C : V ⟶ W) (H : W ⟶ T)
  (B : U ⟶ W) (θ : B.functor ≅ (R ≫ C).functor)

/-- A specified full source node retains its anchor under fixed input precomposition. -/
theorem fixed_finite_node (M : Node H) :
    FinitePrecomp.node B H ⟨R ≫ C, θ⟩ M =
      reanchor (isoWhiskerRight θ H.functor ≪≫
        Functor.associator R.functor C.functor H.functor)
        (precompNode R (precompNode C M)) := by
  apply Sigma.ext
  · rfl
  apply heq_of_eq
  apply Iso.ext
  apply NatTrans.ext
  funext x
  simp only [FinitePrecomp.node, node, reanchor, precompNode, Iso.trans_hom,
    Iso.refl_hom, Functor.mapIso_hom, Iso.prod_hom, isoWhiskerRight_hom,
    isoWhiskerLeft_hom, Functor.associator_hom_app, NatTrans.comp_app,
    whiskerRight_app, whiskerLeft_app, Category.id_comp, Category.comp_id]
  rfl
end Quillen.Waldhausen.IteratedS.CommonReferenceProduct
