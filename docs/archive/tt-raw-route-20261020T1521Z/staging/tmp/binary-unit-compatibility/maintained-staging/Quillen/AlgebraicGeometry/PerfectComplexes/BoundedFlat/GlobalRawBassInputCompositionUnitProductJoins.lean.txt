private def oldEdge : ⦋1⦌ ⟶ ⦋9⦌ := SimplexCategory.Hom.mk ⟨![2, 7], by decide⟩
private def newEdge : ⦋1⦌ ⟶ ⦋9⦌ := SimplexCategory.Hom.mk ⟨![4, 9], by decide⟩

/-- Entire original unit path at composition time 0, before native evaluation. -/
theorem source_zero_path (s : unitInterval) :
    sourceSurface f g (0, s) = parameter (InputIdentitySeedCarrier.laurentRef (f ≫ g))
      (InputIdentitySeedUnitNodes.nodes (f ≫ g)) (InputIdentitySeedUnitNodes.V s) := by
  change parameter _ (sourceNodes f g) (Quillen.SimplexGrid.parameter 1 2 (edge (k := 1) 0 1 s,
    binaryPath (k := 2) 0 1 2 0)) = _
  rw [Path.source, Quillen.SimplexGrid.parameter_column]
  have hv : InputIdentitySeedUnitNodes.V s = embed oldEdge (edge (k := 1) 0 1 s) :=
    (map_edge oldEdge 0 1 s).symm
  rw [hv]
  change parameter _ _ (Convexity.StdSimplex.map _ _) =
    parameter _ _ (Convexity.StdSimplex.map oldEdge.toOrderHom _)
  exact parameter_eq_of_face_nodes (InputIdentitySeedCarrier.laurentRef (f ≫ g))
    (InputIdentitySeedUnitNodes.nodes (f ≫ g)) (sourceNodes f g) oldEdge
      (Quillen.SimplexGrid.columnFace 1 2 0)
    (by
      intro i
      fin_cases i
      · exact source_zero_lower f g
      · exact source_zero_upper f g) (edge (k := 1) 0 1 s)

/-- Entire original unit path at composition time 1, before native evaluation. -/

    sourceSurface f g (1, s) = mapParameter (InputCompositionTerminalAttach.postNode f g)
      (parameter (InputIdentitySeedCarrier.laurentRef g) (InputIdentitySeedUnitNodes.nodes g)
        (InputIdentitySeedUnitNodes.V s)) := by
  change parameter _ (sourceNodes f g) (Quillen.SimplexGrid.parameter 1 2 (edge (k := 1) 0 1 s,
    binaryPath (k := 2) 0 1 2 1)) = _
  rw [Path.target, Quillen.SimplexGrid.parameter_column]
  rw [mapParameter_parameter]
  have hv : InputIdentitySeedUnitNodes.V s = embed oldEdge (edge (k := 1) 0 1 s) :=
    (map_edge oldEdge 0 1 s).symm
  rw [hv]
  change parameter _ _ (Convexity.StdSimplex.map _ _) =
    parameter _ _ (Convexity.StdSimplex.map oldEdge.toOrderHom _)
  exact parameter_eq_of_face_nodes (InputIdentitySeedCarrier.laurentRef (f ≫ g))
    (fun i ↦ InputCompositionTerminalAttach.postNode f g (InputIdentitySeedUnitNodes.nodes g i))
      (sourceNodes f g) oldEdge (Quillen.SimplexGrid.columnFace 1 2 2)
    (by
      intro i
      fin_cases i
      · exact source_one_lower f g
      · exact source_one_upper f g) (edge (k := 1) 0 1 s)
