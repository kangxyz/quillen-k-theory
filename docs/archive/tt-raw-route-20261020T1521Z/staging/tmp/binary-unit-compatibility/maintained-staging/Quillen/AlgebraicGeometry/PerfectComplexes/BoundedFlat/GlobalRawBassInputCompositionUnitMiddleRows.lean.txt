/-- The actual five-node square is the entire lower face of the chosen unit diagram. -/
theorem lower_nodes {X Y : Scheme.{u}} (g : X ⟶ Y) (i : Fin 5) :
    InputNullCompositionLaurentCarrier.directNodes g i =
      InputIdentitySeedUnitNodes.nodes g
        (InputIdentitySeedUnitSeams.unitLowerFace.toOrderHom i) := by
  have h := InputIdentitySeedUnitSeams.outer_nodes g i
  have hi : InputIdentitySeedUnitSeams.rawOuterFace.toOrderHom i =
      BassH0Diagram.plusFace.toOrderHom (ProjectionReference.sharedIndex i) := by
    fin_cases i <;> rfl
  rw [hi] at h
  exact (InputNullCompositionLaurentCarrier.input_nodes g i).symm.trans h

/-- Both transported original lower middle nodes are exactly node 13 of the Laurent rectangle. -/
theorem middle_lower {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) :
    InputCompositionTerminalAttach.postNode f g (InputIdentitySeedUnitNodes.nodes g 4) =
      InputCompositionTerminalAttach.preNode f g (InputIdentitySeedUnitNodes.nodes f 2) := by
  have hg := lower_nodes g 4
  have hf := lower_nodes f 2
  change InputNullCompositionLaurentCarrier.directNodes g 4 =
    InputIdentitySeedUnitNodes.nodes g 4 at hg
  change InputNullCompositionLaurentCarrier.directNodes f 2 =
    InputIdentitySeedUnitNodes.nodes f 2 at hf
  rw [← hg, ← hf]
  exact (InputNullCompositionLaurentCarrier.first_nodes f g 4).symm.trans
    (InputNullCompositionLaurentCarrier.second_nodes f g 2)

/-- Both actual upper middle nodes agree by the proved full-node unit-shift transport. -/
theorem middle_upper {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) :
    InputCompositionTerminalAttach.postNode f g (InputIdentitySeedUnitNodes.nodes g 9) =
      InputCompositionTerminalAttach.preNode f g (InputIdentitySeedUnitNodes.nodes f 7) := by
  have hg := InputIdentitySeedUnitSeams.five_shift g 4
  have hf := InputIdentitySeedUnitSeams.five_shift f 2
  change BassH0Merge.anchorNode (InputIdentitySeedCarrier.commonUnit g)
    (InputIdentitySeedUnitNodes.nodes g 4) = InputIdentitySeedUnitNodes.nodes g 9 at hg
  change BassH0Merge.anchorNode (InputIdentitySeedCarrier.commonUnit f)
    (InputIdentitySeedUnitNodes.nodes f 2) = InputIdentitySeedUnitNodes.nodes f 7 at hf
  rw [← hg, ← hf, ← InputCompositionUnitShift.post_shift,
    ← InputCompositionUnitShift.pre_shift, middle_lower]

private def firstEdge : ⦋1⦌ ⟶ ⦋9⦌ := SimplexCategory.Hom.mk ⟨![4, 9], by decide⟩
private def secondEdge : ⦋1⦌ ⟶ ⦋9⦌ := SimplexCategory.Hom.mk ⟨![2, 7], by decide⟩

/-- The entire transported middle unit path is identical before native evaluation. -/
theorem middle_path {X Y Z : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (s : unitInterval) :
    mapParameter (InputCompositionTerminalAttach.postNode f g)
      (parameter (InputIdentitySeedCarrier.laurentRef g) (InputIdentitySeedUnitNodes.nodes g)
        (InputIdentitySeedUnitNodes.W s)) =
    mapParameter (InputCompositionTerminalAttach.preNode f g)
      (parameter (InputIdentitySeedCarrier.laurentRef f) (InputIdentitySeedUnitNodes.nodes f)
        (InputIdentitySeedUnitNodes.V s)) := by
  rw [mapParameter_parameter, mapParameter_parameter]
  have h1 : InputIdentitySeedUnitNodes.W s = embed firstEdge (edge (k := 1) 0 1 s) :=
    (map_edge firstEdge 0 1 s).symm
  have h2 : InputIdentitySeedUnitNodes.V s = embed secondEdge (edge (k := 1) 0 1 s) :=
    (map_edge secondEdge 0 1 s).symm
  rw [h1, h2]
  change parameter _ _ (Convexity.StdSimplex.map firstEdge.toOrderHom _) =
    parameter _ _ (Convexity.StdSimplex.map secondEdge.toOrderHom _)
  rw [parameter_face, parameter_face]
  apply congrArg (fun N : Fin 2 → Node (InputIdentitySeedCarrier.laurentRef (f ≫ g)) ↦
    parameter (InputIdentitySeedCarrier.laurentRef (f ≫ g)) N (edge (k := 1) 0 1 s))
  funext i
  fin_cases i
  · exact middle_lower f g
  · exact middle_upper f g
