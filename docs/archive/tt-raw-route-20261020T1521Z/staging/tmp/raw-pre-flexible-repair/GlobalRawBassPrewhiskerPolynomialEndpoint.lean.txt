private theorem associator_reanchor {U V W T : Model.{u}}
    (A : U ⟶ V) (E : V ⟶ W) (J : W ⟶ T) (N : Node (E ≫ J)) :
    reanchor (Functor.associator A.functor E.functor J.functor) (precompNode A N) =
      (precompNode A N : Node ((A ≫ E) ≫ J)) := by
  apply Sigma.ext
  · rfl
  apply heq_of_eq
  apply Iso.ext
  ext x
  simp only [precompNode, reanchor, Iso.trans_hom, isoWhiskerLeft_hom, NatTrans.comp_app,
    associator_hom_app, comp_obj, whiskerLeft_app]
  exact Category.id_comp (N.2.hom.app (A.functor.obj x))

private theorem parameter_square
    {U₁ V₁ U₂ V₂ U₃ V₃ U₄ V₄ : Model.{u}}
    {F₁ : U₁ ⟶ V₁} {F₂ : U₂ ⟶ V₂} {F₃ : U₃ ⟶ V₃} {F₄ : U₄ ⟶ V₄}
    (a : Node F₁ → Node F₂) (b : Node F₂ → Node F₄)
    (c : Node F₁ → Node F₃) (d : Node F₃ → Node F₄)
    (h : ∀ N, b (a N) = d (c N)) (y : |parameterSSet F₁|) :
    mapParameter b (mapParameter a y) = mapParameter d (mapParameter c y) := by
  obtain ⟨p, N, w, rfl⟩ := exists_parameter F₁ y
  rw [mapParameter_parameter, mapParameter_parameter, mapParameter_parameter,
    mapParameter_parameter]
  exact congrArg (fun ns ↦ parameter F₄ ns w) (funext fun i ↦ h (N i))

local notation "S" => NativeSquareFiveSimplex.source (polynomial.map c) (projection.app W)
local notation "M" => NativeSquareFiveSimplex.nodes (polynomial.map c) (projection.app W)

local notation "P" => InputCompositionPolynomialScalar.polynomialReference f g
local notation "E" => InputCompositionTopPieces.baseSource f g
local notation "A" => BassH0Merge.normalizedMap c
local notation "J" => ProjectionReference.projectionMap X

/-- The entire polynomial parameter at node four agrees with the actual fixed-source map. -/
theorem terminal_polynomial_parameter (y : |parameterSSet P|) :
    mapParameter (reanchor (projectionSeamAnchor f g c))
      (FinitePrecomp.parameter S M P (y, Convexity.StdSimplex.single 4)) =
    mapParameter (precompNode A)
      (mapParameter (InputCompositionProjectionAttachment.polynomialNode f g) y) := by
  have hv := FiniteParameter.parameter_vertex P (S ≫ P)
    (fun N i ↦ FinitePrecomp.node S P (M i) N) 4 y
  exact (congrArg (mapParameter (reanchor (projectionSeamAnchor f g c))) hv).trans
    (parameter_square (fun N ↦ FinitePrecomp.node S P (M 4) N)
      (reanchor (projectionSeamAnchor f g c))
      (InputCompositionProjectionAttachment.polynomialNode f g) (precompNode A)
      (fun N ↦ (terminal_reanchored_node f g c N).trans
        (associator_reanchor A E J (InputCompositionProjectionAttachment.polynomialNode f g N))) y)
