module
public import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalRawBassPrewhiskerFamily
/-! A downstream client retains original raw prewhiskering and independent loop/structure clocks. -/
@[expose] public section
noncomputable section
set_option autoImplicit false
open CategoryTheory AlgebraicGeometry Quillen.PerfectComplexes.GlobalRawBass
open Quillen.PerfectComplexes Quillen.LoopPrespectrum
open scoped unitInterval
universe u
namespace Consumer.GlobalRawBassPrewhisker
open InputCompositionPrewhisker
variable {X Y Z W : Scheme.{u}} (f : X ⟶ Y) (g : Y ⟶ Z) (c : Z ⟶ W)

/-- The independent loop coordinate recovers the entire original chosen raw family. -/
theorem original_loop (n : ℕ) (r t : I)
    (x : (parameterLooping (mappingCone (comparison W))).space n) :
    Path.toContinuousMap ((rawPreFamily f g c).looped n (r, x)) t =
      (InputComposition.coneHomotopy f g).level n
        (r, (coneMap c).map n (Path.toContinuousMap x t)) :=
  rawPreFamily_desc f g c n r (Path.toContinuousMap x t)

/-- Structure time remains independent of both composition and loop coordinates. -/
theorem independent_structure (n : ℕ) (r z : I)
    (x : (parameterLooping (mappingCone (comparison W))).space n) :
    (rawPreFamily f g c).looped (n + 1)
      (r, (parameterLooping (mappingCone (comparison W))).structureMap n x z) =
        (parameterLooping (mappingCone (comparison X))).structureMap n
          ((rawPreFamily f g c).looped n (r, x)) z :=
  (rawPreFamily f g c).looped_structure n r x z
end Consumer.GlobalRawBassPrewhisker
