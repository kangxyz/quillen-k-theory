/-!
The exact full terminal reference of the original global H0 diagram. Its actual geometric
  outer-projection anchor and binary normalization total identify the complete original terminal
  node.
-/
@[expose] public section
noncomputable section
set_option autoImplicit false
set_option backward.isDefEq.respectTransparency false
set_option backward.defeqAttrib.useBackward true
open CategoryTheory CategoryTheory.Functor AlgebraicGeometry
open AlgebraicGeometry.Scheme.PolynomialLaurent.GlobalIndex
open Quillen.Waldhausen.IteratedS
universe u
namespace Quillen.PerfectComplexes.GlobalRawBass.ProjectionReference.Target
open BassH0Merge CommonReferenceNerve
variable {X Y : Scheme.{u}} (f : X ⟶ Y)

/-- The actual terminal H0 reference, keeping the projection equalities and normalization. -/
def terminalBaseAnchor : (InputDescSimplex.reference f).functor ≅
    (InputCompositionTerminalAttach.arrow f ≫ InputIdentityProjection.source X).functor :=
  eqToIso (Quillen.PerfectComplexes.ProjectionReference.outerAnchor
    (projection.app Y) (plus.app Y) (projection.app X) (plus.app X)
    f (polynomial.map f) (laurent.map f) (projection.naturality f) (plus.naturality f)
    (laurentProjection.app Y) (plus.app X ≫ projection.app X) (plus_projection Y) rfl) ≪≫
      BinaryNormalizationSimplex.total (plus.app X ≫ projection.app X) f

/-- The normalized original terminal vertex has precisely this full reference. -/
theorem terminal_normal_node :
    InputDescSimplex.chartNodes f plus plus_projection 15 =
      (⟨InputCompositionTerminalAttach.arrow f ≫ InputIdentityProjection.source X,
        terminalBaseAnchor f⟩ : CommonReferenceNerve.Node (InputDescSimplex.reference f)) := by
  apply Sigma.ext (Quillen.PerfectComplexes.ProjectionReference.target_normal_map
    (projection.app Y) (plus.app Y) (projection.app X) (plus.app X)
    f (polynomial.map f) (laurent.map f) (projection.naturality f) (plus.naturality f)
    (laurentProjection.app Y) (plus_projection Y))
  apply heq_of_eq
  exact Quillen.PerfectComplexes.ProjectionReference.target_normal_reference
    (projection.app Y) (plus.app Y) (projection.app X) (plus.app X)
    f (polynomial.map f) (laurent.map f) (projection.naturality f) (plus.naturality f)
    (laurentProjection.app Y) (plus_projection Y)
end Quillen.PerfectComplexes.GlobalRawBass.ProjectionReference.Target
