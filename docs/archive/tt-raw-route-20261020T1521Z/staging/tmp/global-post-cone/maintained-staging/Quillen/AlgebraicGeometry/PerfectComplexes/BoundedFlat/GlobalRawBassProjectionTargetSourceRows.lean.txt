# The complete source row of the original target comparison

The initial H0 row is the unchanged five-node Bass projection path postcomposed with the

/-- Equality of the actual composite transports the entire original binary total. -/
theorem source_total_transport {A B C D : Scheme.{u}}
    (p : B ⟶ A) (q : C ⟶ B) (c : D ⟶ C) (r : C ⟶ A) (hr : q ≫ p = r) :
    eqToIso (ProjectionReference.anchor p q c r hr) ≪≫
        BinaryNormalizationSimplex.total c (q ≫ p) =
      BinaryNormalizationSimplex.total c r ≪≫
        isoWhiskerRight (eqToIso (congrArg (fun k ↦ (normalizedMap k).functor) hr.symm))
          (normalizedMap c).functor := by
  subst r
  simp only [eqToIso_refl, Iso.refl_trans, isoWhiskerRight_refl]
  exact (Iso.trans_refl (BinaryNormalizationSimplex.total c (q ≫ p))).symm

variable {X Y : Scheme.{u}} (f : X ⟶ Y)
/-- The literal Laurent coefficient map at the target. -/
abbrev sourcePost := InputCompositionTerminalAttach.arrow (laurent.map f)

/-- The explicit source projection transport for either original chart. -/
def sourceChartIso (α : laurent ⟶ polynomial)
    (ha : ∀ Z, α.app Z ≫ projection.app Z = laurentProjection.app Z) :
    (normalizedMap (laurentProjection.app Y)).functor ≅
      (BinaryNormalizationSimplex.sourceMap (α.app Y)
        (projection.app Y)).functor :=
  eqToIso (congrArg (fun k ↦ (normalizedMap k).functor) (ha Y).symm)

/-- The complete source-column anchor retains both its total and projection equality. -/
def sourceRowAnchor (α : laurent ⟶ polynomial)
    (ha : ∀ Z, α.app Z ≫ projection.app Z = laurentProjection.app Z) :
    (InputDescSimplex.reference f).functor ≅
      (BinaryNormalizationSimplex.sourceMap (α.app Y)
        (projection.app Y) ≫ sourcePost f).functor :=
  BinaryNormalizationSimplex.total
    (laurent.map f) (laurentProjection.app Y) ≪≫
      isoWhiskerRight (sourceChartIso α ha) (sourcePost f).functor

/-- The literal positive projection source transport. -/
abbrev projectionReferenceIso (Y : Scheme.{u}) := sourceChartIso (Y := Y) plus plus_projection

/-- Original source-column reference with the actual positive projection source. -/
abbrev h0ProjectionIso := sourceRowAnchor f plus plus_projection

/-- All three full source nodes are the actual binary column with its normalized anchor. -/
theorem source_chart_nodes
    (α : laurent ⟶ polynomial)
    (ha : ∀ Z, α.app Z ≫ projection.app Z = laurentProjection.app Z)
    (i : Fin 3) :
    InputDescSimplex.chartNodes f α ha (rectangleLeftIndex i) =
      postReferenceNode (InputDescSimplex.reference f) (sourcePost f) (sourceRowAnchor f α ha)
        (binaryNodes (α.app Y) (projection.app Y) i) := by
  rw [show InputDescSimplex.chartNodes f α ha (rectangleLeftIndex i) = _ from
    native_source_column (projection.app Y) (α.app Y) (projection.app X) (α.app X)
      f (polynomial.map f) (laurent.map f) (projection.naturality f) (α.naturality f)
      (laurentProjection.app Y) (ha Y) i]
  change (⟨_, (eqToIso (ProjectionReference.anchor _ _ _ _ (ha Y)) ≪≫
    BinaryNormalizationSimplex.total _ _) ≪≫ _⟩ : CommonReferenceNerve.Node _) = _
  rw [source_total_transport _ _ _ _ (ha Y)]
  rfl

/-- The positive anchor is exactly the public H0-to-source projection reference. -/
theorem plus_source_anchor :
    sourceRowAnchor f (plus) (plus_projection) =
      h0ProjectionIso f := rfl

/-- The inverse source uses precisely the original positive-to-inverse source transport. -/
theorem minus_source_anchor :
    sourceRowAnchor f (minus) (minus_projection) =
      h0ProjectionIso f ≪≫
        isoWhiskerRight (InputIdentityProjection.sourceIso Y) (sourcePost f).functor := by
  have h : sourceChartIso (minus) (minus_projection) =
      projectionReferenceIso Y ≪≫ InputIdentityProjection.sourceIso Y := by
    apply Iso.ext
    simp only [sourceChartIso, projectionReferenceIso, InputIdentityProjection.sourceIso,
      Iso.trans_hom, eqToIso.hom, eqToHom_trans]
  dsimp only [sourceRowAnchor, h0ProjectionIso, sourcePost]
  rw [h, isoWhiskerRight_trans, Iso.trans_assoc]

/-- The five source nodes retain the exact shared center and both separate intermediates. -/
def sourceRow : Fin 5 → Fin 31 := ![2, 3, 11, 14, 22]

/-- The complete initial H0 row equals the unchanged five-node Bass projection family. -/
theorem source_nodes (i : Fin 5) :
    InputDescSimplex.nodes f (sourceRow i) =
      postReferenceNode (InputDescSimplex.reference f) (sourcePost f) (h0ProjectionIso f)
        (InputIdentityProjection.nodes Y i) := by
  fin_cases i
  · exact (InputDescSimplex.plus_nodes f (rectangleLeftIndex 0)).trans
      ((source_chart_nodes f (plus) (plus_projection) 0).trans
        (by rw [plus_source_anchor]; rfl))
  · exact (InputDescSimplex.plus_nodes f (rectangleLeftIndex 1)).trans
      ((source_chart_nodes f (plus) (plus_projection) 1).trans
        (by rw [plus_source_anchor]; rfl))
  · exact (InputDescSimplex.plus_nodes f (rectangleLeftIndex 2)).trans
      ((source_chart_nodes f (plus) (plus_projection) 2).trans
        (by rw [plus_source_anchor]; rfl))
  · refine (InputDescSimplex.minus_nodes f (rectangleLeftIndex 1)).trans
      ((source_chart_nodes f (minus) (minus_projection) 1).trans ?_)
    rw [minus_source_anchor]
    exact (post_anchor_node (InputDescSimplex.reference f) (sourcePost f)
      (h0ProjectionIso f) (InputIdentityProjection.sourceIso Y)
      (binaryNodes (minus.app Y) (projection.app Y) 1)).symm
  · refine (InputDescSimplex.minus_nodes f (rectangleLeftIndex 2)).trans
      ((source_chart_nodes f (minus) (minus_projection) 2).trans ?_)
    rw [minus_source_anchor]
    exact (post_anchor_node (InputDescSimplex.reference f) (sourcePost f)
      (h0ProjectionIso f) (InputIdentityProjection.sourceIso Y)
      (binaryNodes (minus.app Y) (projection.app Y) 2)).symm
