import Quillen.AlgebraicGeometry.PerfectComplexes.BoundedFlat.GlobalSupportedRawInputBinarySurface

/-! Regression checks against both actual original comparison tracks. -/

open CategoryTheory AlgebraicGeometry
open Quillen.Waldhausen.IteratedS
open Quillen.LoopPrespectrum.Homotopy.RelHomotopy
open Quillen.PerfectComplexes.GlobalSupportedRawBass
open Quillen.PerfectComplexes.GlobalSupportedRawBass.InputBinary
open scoped unitInterval
universe u
variable {X Y Z : Scheme.OpenPairs.{u}} (f : X ⟶ Y) (g : Y ⟶ Z)
-- Independent checks of both original chosen comparison faces.
example (n : ℕ) (s : I) (x : (rawSpectrum Z).space n) :
    surfaceValue (compositionComparison f g) n 0 s x =
      (normalizedCompositionHomotopy f g).level n (s, x) :=
  surfaceValue_zero (compositionComparison f g) n s x

example (n : ℕ) (s : I) (x : (rawSpectrum Z).space n) :
    surfaceValue (compositionComparison f g) n 1 s x =
      (compositionHomotopy f g).level n (s, x) :=
  surfaceValue_one (compositionComparison f g) n s x
